```python
class ContractRiskLevel:
    """合约风险等级"""
    
    SAFE = "SAFE"           # ✅ 低风险：审计通过、无升级、无漏洞
    MEDIUM = "MEDIUM"       # ⚠️ 中风险：有升级权限但有时间锁
    HIGH = "HIGH"           # 🚨 高风险：无限升级权限或未审计
    CRITICAL = "CRITICAL"   # ☠️ 极高风险：已知漏洞或已发生攻击

def classify_contract_risk(contract_address: str, web3) -> dict:
    """
    分类智能合约风险
    
    检查项：
    1. 审计状态
    2. 升级权限
    3. 管理员权限
    4. 合约复杂度
    5. 历史漏洞
    """
    checks = {}
    
    # 1. 审计检查
    audit_status = check_audit(contract_address)
    checks["audit"] = audit_status
    
    # 2. 升级权限
    proxy_pattern = detect_proxy_pattern(contract_address, web3)
    checks["proxy"] = proxy_pattern
    
    # 3. 管理员权限
    admin_perms = check_admin_permissions(contract_address, web3)
    checks["admin"] = admin_perms
    
    # 4. 暂停开关
    pause开关 = check_pause_switch(contract_address, web3)
    checks["pause_switch"] = pause开关
    
    # 5. 代币approve风险
    approve_risk = check_approve_risk(contract_address, web3)
    checks["approve_risk"] = approve_risk
    
    # 综合评分
    risk_score = calculate_risk_score(checks)
    
    return {
        "contract": contract_address,
        "checks": checks,
        "risk_score": risk_score,
        "risk_level": get_risk_level(risk_score),
        "recommendations": get_risk_recommendations(checks)
    }

def get_risk_level(score: float) -> str:
    if score >= 80:
        return ContractRiskLevel.CRITICAL
    elif score >= 60:
        return ContractRiskLevel.HIGH
    elif score >= 30:
        return ContractRiskLevel.MEDIUM
    else:
        return ContractRiskLevel.SAFE

def calculate_risk_score(checks: dict) -> float:
    """
    计算风险评分（0-100，越高越危险）
    """
    score = 0
    
    # 审计（40分）
    if not checks["audit"]["has_audit"]:
        score += 40
    elif checks["audit"]["critical_issues"] > 0:
        score += 30
    elif checks["audit"]["high_issues"] > 0:
        score += 15
    
    # 代理模式（20分）
    if checks["proxy"]["has_proxy"]:
        score += 20
        if not checks["proxy"]["has_timelock"]:
            score += 10
    
    # 管理员权限（20分）
    if checks["admin"]["can_mint"]:
        score += 10
    if checks["admin"]["can_freeze"]:
        score += 5
    if checks["admin"]["can_blacklist"]:
        score += 5
    
    # approve风险（20分）
    if checks["approve_risk"]["has_unlimited_approve"]:
        score += 15
    if checks["approve_risk"]["known_exploitable"]:
        score += 20
    
    return min(score, 100)
```

```python
    # 评分公式
    # Critical = -30

        1 * 15 +   # High
        3 * 5 +    # Medium
        8 * 2 +    # Low
        15 * 0.5   # Informational
    )
```

```python
class SecurityMonitor:
    """链上安全监控系统"""
    
    def __init__(self, web3, alerts_config: dict):
        self.w3 = web3
        self.alerts = alerts_config  # 告警配置
    
    def monitor_contract_events(self, contract_address: str) -> list:
        """
        监控合约关键事件
        """
        events_to_monitor = [
            "AdminChanged",
            "Upgraded",
            "Paused",
            "Unpaused",
            "RoleGranted",
            "RoleRevoked",
            "Approve",
            "Transfer"
        ]
        
        alerts = []
        
        # 获取最新区块
        latest_block = self.w3.eth.block_number
        
        for event_name in events_to_monitor:
            try:
                # 获取过去 10 个区块的该事件
                events = self.get_past_events(
                    contract_address,
                    event_name,
                    from_block=latest_block - 10,
                    to_block=latest_block
                )
                
                if events:
                    alerts.append({
                        "event": event_name,
                        "count": len(events),
                        "latest_block": events[0]["blockNumber"],
                        "severity": self._get_event_severity(event_name)
                    })
            except Exception as e:
                print(f"Error monitoring {event_name}: {e}")
        
        return alerts
    
    def _get_event_severity(self, event_name: str) -> str:
        """事件严重性分级"""
        critical_events = ["Upgraded", "Paused", "AdminChanged"]
        high_events = ["RoleGranted", "RoleRevoked"]
        
        if event_name in critical_events:
            return "CRITICAL"
        elif event_name in high_events:
            return "HIGH"
        else:
            return "MEDIUM"
    
    def monitor_large_transfers(self, token_address: str, threshold_usd=100000) -> list:
        """
        监控大额转账
        链上监控重要部分
        """
        # 获取 Transfer 事件
        latest_block = self.w3.eth.block_number
        
        events = self.get_past_events(
            token_address,
            "Transfer",
            from_block=latest_block - 100,
            to_block=latest_block
        )
        
        large_transfers = []
        for event in events:
            value_usd = self._calculate_transfer_value(event, token_address)
            if value_usd >= threshold_usd:
                large_transfers.append({
                    "hash": event["transactionHash"].hex(),
                    "from": event["args"]["from"],
                    "to": event["args"]["to"],
                    "value_usd": value_usd,
                    "block": event["blockNumber"],
                    "timestamp": self._get_block_timestamp(event["blockNumber"])
                })
        
        return sorted(large_transfers, key=lambda x: x["value_usd"], reverse=True)
    
    def monitor_approval_changes(self, address: str, spender_threshold_usd=10000) -> list:
        """
        监控授权变更
        防止无限授权和授权给恶意合约
        """
        latest_block = self.w3.eth.block_number
        
        events = self.get_past_events(
            None,  # 代币地址需指定
            "Approval",
            from_block=latest_block - 1000,
            to_block=latest_block,
            argument_filters={"owner": address}
        )
        
        risky_approvals = []
        for event in events:
            approved_amount = event["args"]["value"]
            is_unlimited = approved_amount == 2**256 - 1
            value_usd = self._calculate_approval_value(event, approved_amount)
            
            # 风险评估
            if is_unlimited or value_usd >= spender_threshold_usd:
                # 检查 spender 是否可信
                spender_contract = event["args"]["spender"]
                is_contract = self._is_contract(spender_contract)
                is_verified = self._is_verified_contract(spender_contract)
                
                risky_approvals.append({
                    "spender": spender_contract,
                    "is_unlimited": is_unlimited,
                    "approved_value_usd": value_usd,
                    "is_contract": is_contract,
                    "is_verified": is_verified,
                    "risk_level": "HIGH" if (is_unlimited and not is_verified) else "MEDIUM",
                    "recommendation": "撤销授权" if is_unlimited and not is_verified else "监控"
                })
        
        return risky_approvals
    
    def detect_suspicious_activity(self, contract_address: str) -> dict:
        """
        检测可疑活动
        """
        latest_block = self.w3.eth.block_number
        
        # 检测指标
        indicators = {}
        
        # 1. 异常大额转账
        large_txs = self.monitor_large_transfers(contract_address, threshold_usd=50000)
        indicators["large_transfers"] = {
            "count": len(large_txs),
            "total_usd": sum(tx["value_usd"] for tx in large_txs),
            "risk": "HIGH" if len(large_txs) > 10 else "MEDIUM"
        }
        
        # 2. admin 权限变更
        admin_changes = self.get_past_events(
            contract_address, "RoleGranted",
            from_block=latest_block - 100,
            to_block=latest_block
        )
        indicators["admin_changes"] = {
            "count": len(admin_changes),
            "risk": "HIGH" if len(admin_changes) > 0 else "LOW"
        }
        
        # 3. 合约暂停
        pause_events = self.get_past_events(
            contract_address, "Paused",
            from_block=latest_block - 1000,
            to_block=latest_block
        )
        indicators["contract_paused"] = {
            "count": len(pause_events),
            "risk": "HIGH" if len(pause_events) > 0 else "LOW"
        }
        
        # 综合判断
        overall_risk = max(
            self._risk_to_score(indicators[k]["risk"]) 
            for k in indicators
        )
        
        return {
            "indicators": indicators,
            "overall_risk": self._score_to_risk(overall_risk),
            "alerts": self._generate_alerts(indicators)
        }
```

```python
def detect_rug_pull_risk(token_address: str) -> dict:
    """
    检测代币的 Rug Pull 风险
    
    关键指标：
    1. 流动性锁定情况
    2. 团队代币比例和锁仓
    3. 合约可升级性
    4. 异常交易模式
    5. 社交媒体信号
    """
    risk_indicators = {}
    
    # 1. 流动性锁定检查
    liquidity_lock = check_liquidity_lock(token_address)
    risk_indicators["liquidity_lock"] = {
        "is_locked": liquidity_lock["locked"],
        "lock_until": liquidity_lock.get("lock_until"),
        "locked_percentage": liquidity_lock.get("locked_pct", 0),
        "risk": "LOW" if liquidity_lock["locked"] else "CRITICAL"
    }
    
    # 2. 代币分配检查
    token_allocation = check_token_allocation(token_address)
    risk_indicators["token_allocation"] = {
        "team_allocation": token_allocation.get("team_pct", 0),
        "marketing_allocation": token_allocation.get("marketing_pct", 0),
        "is_team_locked": token_allocation.get("team_locked", False),
        "risk": "HIGH" if token_allocation.get("team_pct", 0) > 20 else "LOW"
    }
    
    # 3. 合约升级风险
    upgrade_risk = check_upgrade_risk(token_address)
    risk_indicators["upgrade_risk"] = {
        "has_proxy": upgrade_risk["has_proxy"],
        "has_timelock": upgrade_risk["has_timelock"],
        "admin_can_upgrade": upgrade_risk["admin_can_upgrade"],
        "risk": "HIGH" if upgrade_risk["admin_can_upgrade"] and not upgrade_risk["has_timelock"] else "LOW"
    }
    
    # 4. 交易历史分析
    trading_pattern = analyze_trading_pattern(token_address)
    risk_indicators["trading_pattern"] = {
        "sell_pressure_24h": trading_pattern["sell_pressure"],
        "volume_spike": trading_pattern["volume_spike"],
        "smart_money_outflow": trading_pattern["smart_money_outflow"],
        "risk": "HIGH" if trading_pattern["volume_spike"] > 5 else "MEDIUM"
    }
    
    # 综合评分
    overall_risk = (
        (100 if risk_indicators["liquidity_lock"]["risk"] == "CRITICAL" else
         60 if risk_indicators["liquidity_lock"]["risk"] == "HIGH" else 20) * 0.3 +
        (80 if risk_indicators["token_allocation"]["risk"] == "HIGH" else 20) * 0.2 +
        (80 if risk_indicators["upgrade_risk"]["risk"] == "HIGH" else 20) * 0.2 +
        (80 if risk_indicators["trading_pattern"]["risk"] == "HIGH" else 40) * 0.3
    )
    
    return {
        "token": token_address,
        "risk_indicators": risk_indicators,
        "overall_risk_score": overall_risk,
        "overall_risk_level": "CRITICAL" if overall_risk >= 70 else "HIGH" if overall_risk >= 50 else "MEDIUM" if overall_risk >= 30 else "LOW",
        "recommendations": get_rug_risk_recommendations(risk_indicators)
    }

def check_liquidity_lock(token_address: str) -> dict:
    """
    检查流动性锁定
    """
    # 检查 LP 代币是否被锁定
    # 常用锁定平台：Unicrypt, Team Finance, DXsale
    
    # 获取 LP 合约地址（需要先找到 LP）
    lp_address = find_lp_address(token_address)
    
    if not lp_address:
        return {"locked": False, "reason": "No LP found"}
    
    # 检查锁定状态
    lock_info = check_unicrypt_lock(lp_address)
    
    if lock_info:
        return {
            "locked": lock_info["is_locked"],
            "lock_until": lock_info.get("unlock_time"),
            "locked_pct": lock_info.get("locked_pct", 0),
            "locker_address": lock_info.get("locker")
        }
    
    return {"locked": False, "reason": "Not verified"}
```

```python
    institutional_flows = categorize_institutional(transfers)
    smart_money = track_smart_money_wallets(transfers)
    
    return {
        "total_whale_txs_24h": len(transfers),
        "total_volume_24h_usd": sum(t["value_usd"] for t in transfers),
        "exchange_flows": exchange_flows,
        "institutional_flows": institutional_flows,
        "smart_money_signals": smart_money,
        "net_flow": calculate_net_flow(transfer)
    }

def calculate_net_flow(transfers: list) -> dict:
    """
    计算净流量
    净流入 = 到期地址（持有地址、交易所充值地址）
    净流出 = 从期地址（个人钱包、交易所提现地址）
    """
    inflow = 0
    outflow = 0
    
    for tx in transfers:
        if is_inflow_address(tx["to"]):
            inflow += tx["value_usd"]
        if is_outflow_address(tx["from"]):
            outflow += tx["value_usd"]
    
    net = inflow - outflow
    
    return {
        "inflow_usd": inflow,
        "outflow_usd": outflow,
        "net_flow_usd": net,
        "signal": "bullish" if net > 1000000 else "neutral"
    }
```

## 五、闪电贷攻击防御

### 闪电贷攻击原理

```
标准闪电贷攻击步骤：
1. 从 Aave / dYdX 借出大量 Token X
2. 在 DEX 上操控价格（大量买入）
3. 在另一个协议触发价格依赖逻辑（清算 / 交换）
4. 归还借款
5. 利润 = 操作收益 - 借款费用

典型攻击场景：
- Beanstalk：借 ETH → 操控治理投票 → 窃取 $182M
- Mango Markets：借 USDC → 操控 MNGO 价格 → 提取 $117M
- Euler Finance：借 ETH → 恶意提案 → $197M
```

```python
    if has_governance(protocol_address):
        if not has_timelock(protocol_address):
```

```python
def pre_trade_security_check(
    token_address: str,
    amount_usd: float,
    protocol_address: str = None
) -> dict:
    """
    交易前安全检查
    
    检查清单：
    1. 合约审计状态
    2. Rug Pull 风险
    3. 闪电贷漏洞
    4. 授权风险
    5. 合约暂停状态
    """
    checks = {}
    
    # 1. 合约审计
    audit = check_audit(token_address)
    checks["audit"] = {
        "has_audit": audit["has_audit"],
        "passed": audit["passes"] if audit["has_audit"] else False,
        "risk": "LOW" if audit["has_audit"] and audit["passes"] else "HIGH"
    }
    
    # 2. Rug Pull 风险
    rug_risk = detect_rug_pull_risk(token_address)
    checks["rug_risk"] = {
        "score": rug_risk["overall_risk_score"],
        "level": rug_risk["overall_risk_level"],
        "risk": "LOW" if rug_risk["overall_risk_level"] == "LOW" else "HIGH"
    }
    
    # 3. 合约暂停
    if protocol_address:
        is_paused = check_pause_status(protocol_address)
        checks["pause"] = {
            "is_paused": is_paused,
            "risk": "HIGH" if is_paused else "LOW"
        }
    
    # 4. 授权风险
    approve_risk = check_user_approvals()
    checks["approve"] = {
        "unlimited_approvals": approve_risk["unlimited_approvals"],
        "risk": "MEDIUM" if approve_risk["unlimited_approvals"] else "LOW"
    }
    
    # 5. 流动性检查
    liquidity = get_token_liquidity(token_address)
    checks["liquidity"] = {
        "tvl_usd": liquidity,
        "sufficient": liquidity >= amount_usd * 10,  # 至少 10 倍于交易额
        "risk": "LOW" if liquidity >= amount_usd * 10 else "HIGH"
    }
    
    # 综合判断
    risk_factors = [checks[k]["risk"] for k in checks]
    overall_risk = "HIGH" if "HIGH" in risk_factors else "MEDIUM" if "MEDIUM" in risk_factors else "LOW"
    
    return {
        "token": token_address,
        "checks": checks,
        "overall_risk": overall_risk,
        "trade_recommended": overall_risk != "HIGH",
        "warnings": generate_warnings(checks),
        "action": "PROCEED" if overall_risk == "LOW" else "CAUTION" if overall_risk == "MEDIUM" else "ABORT"
    }
```
