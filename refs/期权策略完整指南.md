## 一、期权基础概念

### 期权定价模型

```python
from scipy.stats import norm
import numpy as np

class BlackScholes:
    """Black-Scholes 期权定价模型"""
    
    @staticmethod
    def call_price(S, K, T, r, sigma):
        """
        计算 Call 期权价格
        
        S = 标的资产价格
        K = 行权价
        T = 到期时间（年）
        r = 无风险利率
        sigma = 波动率
        """
        if T <= 0:
            return max(S - K, 0)
        
        d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        call = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
        return call
    
    @staticmethod
    def put_price(S, K, T, r, sigma):
        """计算 Put 期权价格"""
        if T <= 0:
            return max(K - S, 0)
        
        d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        put = K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
        return put
    
    @staticmethod
    def implied_volatility(market_price, S, K, T, r, option_type="call"):
        """
        计算隐含波动率 (IV)
        通过市场价格反推波动率
        """
        from scipy.optimize import brentq
        
        def objective(sigma):
            if option_type == "call":
                price = BlackScholes.call_price(S, K, T, r, sigma)
            else:
                price = BlackScholes.put_price(S, K, T, r, sigma)
            return price - market_price
        
        try:
            iv = brentq(objective, 0.001, 5.0)
            return iv
        except:
            return None

def get_greeks(S, K, T, r, sigma, option_type="call"):
    """
    计算期权希腊字母
    Delta, Gamma, Vega, Theta, Rho
    """
    if T <= 0:
        return {"delta": 1 if option_type == "call" and S > K else -1 if option_type == "put" and S < K else 0,
                "gamma": 0, "vega": 0, "theta": 0, "rho": 0}
    
    d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    if option_type == "call":
        delta = norm.cdf(d1)
        rho = K * T * np.exp(-r * T) * norm.cdf(d2) / 100
    else:
        delta = norm.cdf(d1) - 1
        rho = -K * T * np.exp(-r * T) * norm.cdf(-d2) / 100
    
    gamma = norm.pdf(d1) / (S * sigma * np.sqrt(T))
    vega = S * norm.pdf(d1) * np.sqrt(T) / 100  # 每 1% 波动率变化
    theta = (-S * norm.pdf(d1) * sigma / (2 * np.sqrt(T))
             - r * K * np.exp(-r * T) * norm.cdf(d2 if option_type == "call" else -d2)) / 365
    
    return {
        "delta": delta,   # 标的价格变化对期权价格的影响
        "gamma": gamma,   # Delta 对标的价格变化的敏感度
        "vega": vega,     # 波动率变化对期权价格的影响
        "theta": theta,   # 时间衰减（每天损失的价值）
        "rho": rho        # 利率变化对期权价格的影响
    }
```

```python
    """判断期权价值状态"""

        return "ITM"  # 实值
    elif ratio < 0.95:

def calculate_intrinsic_value(S, K, option_type):
    """计算期权的内在价值"""
    if option_type == "call":
        return max(S - K, 0)
    else:
        return max(K - S, 0)

def calculate_time_value(S, K, T, r, sigma, market_price, option_type):
    """计算期权的时间价值"""
    intrinsic = calculate_intrinsic_value(S, K, option_type)
    theoretical = BlackScholes.call_price(S, K, T, r, sigma) if option_type == "call" \
                  else BlackScholes.put_price(S, K, T, r, sigma)
    return max(theoretical - intrinsic, 0)
```

## 二、波动率分析

### 波动率计算

```python
import numpy as np
import pandas as pd

class VolatilityAnalyzer:
    """波动率分析工具"""
    
    @staticmethod
    def historical_volatility(prices: list, window=30) -> float:
        """
        计算历史波动率 (HV)
        
        公式: HV = sqrt(252 * sum((ln(P_i/P_{i-1}))^2) / (n-1))
        
        prices: 价格序列
        window: 计算窗口（天）
        """
        prices = np.array(prices[-window:])
        log_returns = np.log(prices[1:] / prices[:-1])
        
        hv = np.std(log_returns) * np.sqrt(252)
        return hv
    
    @staticmethod
    def parkinson_volatility(high: list, low: list, window=30) -> float:
        """
        Parkinson 波动率（使用高低价）
        对极端值更敏感，适合快速估计
        """
        high = np.array(high[-window:])
        low = np.array(low[-window:])
        
        hl_ratio = np.log(high / low)
        parkinson = np.sqrt(np.mean(hl_ratio**2) / (4 * np.log(2))) * np.sqrt(252)
        return parkinson
    
    @staticmethod
    def yang_zhang_volatility(open_prices, high_prices, low_prices, close_prices, window=30) -> float:
        """
        Yang-Zhang 波动率（最准确的 HV 估计）
        结合了开盘跳空、夜间波动和日内波动
        """
        n = len(open_prices)
        opens = np.array(open_prices[-window:])
        highs = np.array(high_prices[-window:])
        lows = np.array(low_prices[-window:])
        closes = np.array(close_prices[-window:])
        
        # 夜间波动（从收盘到次日开盘）
        overnight = np.log(opens[1:] / closes[:-1])
        sigma_n = np.std(overnight, ddof=1)**2
        
        # 日内波动（Open to Close）
        open_close = np.log(closes / opens)
        sigma_oc = np.std(open_close, ddof=1)**2
        
        # 跳空波动（High to Low）
        hl = np.log(highs / lows)
        sigma_hl = np.mean(hl**2)
        
        # Yang-Zhang 估计
        k = 0.34 / (1.34 + (n + 1) / (n - 1))
        yz_var = sigma_n + k * sigma_oc + (1 - k) * sigma_hl
        
        return np.sqrt(yz_var * 252)
    
    @staticmethod
    def realized_volatility_5min(minute_prices: list) -> float:
        """
        5分钟收益率计算已实现波动率（RV）
        这是波动率估计的金标准
        """
        returns = np.diff(np.log(minute_prices))
        rv = np.sqrt(np.sum(returns**2) * 252 * 78)  # 每天78个5分钟K线
        return rv
    
    @staticmethod
    def garch_volatility_estimate(returns: list) -> float:
        """
        GARCH(1,1) 波动率估计
        更平滑，能捕捉波动率聚集效应
        """
        returns = np.array(returns)
        
        # 参数初始化
        omega = 0.000001
        alpha = 0.1
        beta = 0.85
        
        # 简化的 GARCH 估计
        # 实际应用中用 arch 库
        variance = np.var(returns)
        
        for r in returns[-30:]:
            variance = omega + alpha * r**2 + beta * variance
        
        return np.sqrt(variance * 252)
    
    @staticmethod
    def iv_hv_ratio(iv: float, hv: float) -> dict:
        """
        IV/HV 比率分析
        IV > HV → 市场预期未来波动 > 历史波动
        IV < HV → 市场预期未来波动 < 历史波动
        """
        ratio = iv / hv if hv > 0 else 0
        
        return {
            "iv": iv,
            "hv": hv,
            "ratio": ratio,
            "signal": "IV 高估" if ratio > 1.2 else "IV 低估" if ratio < 0.8 else "中性",
            "premium": (ratio - 1) * 100,  # 隐含波动率溢价 %
        }

def estimate_iv_rank(current_iv: float, iv_history: list) -> float:
    """
    计算 IV Rank
    当前 IV 在历史 IV 分布中的百分位
    """
    iv_history = np.array(iv_history)
    rank = (current_iv - np.min(iv_history)) / (np.max(iv_history) - np.min(iv_history) + 1e-10)
    return rank * 100

def calculate_volatility_surface(strike_range, expiry_range, base_vol, skew_factor=0.3) -> np.ndarray:
    """
    构建波动率曲面
    考虑 OTM Put 的波动率溢价（Skew）
    """
    surface = np.zeros((len(expiry_range), len(strike_range)))
    
    for i, expiry in enumerate(expiry_range):
        for j, strike in enumerate(strike_range):
            moneyness = np.log(strike / base_vol)  # 简化的 moneyness
            # Put skew: OTM put 的 IV 更高
            skew_adjustment = -skew_factor * moneyness  # 负 moneyness = OTM put
            
            surface[i, j] = base_vol + skew_adjustment * np.sqrt(expiry / 30)
    
    return surface
```

```python
def calculate_portfolio_greeks(positions: list) -> dict:
    """
    计算组合总希腊字母
    positions: [{"type": "call", "strike": 60000, "expiry": "28JUN24", "size": 10, "iv": 0.6}, ...]
    """
    total = {"delta": 0, "gamma": 0, "vega": 0, "theta": 0, "rho": 0}
    
    for pos in positions:
        S = pos["underlying_price"]
        K = pos["strike"]
        T = pos["days_to_expiry"] / 365
        r = 0.05
        sigma = pos["iv"]
        
        greeks = get_greeks(S, K, T, r, sigma, pos["type"])
        
        for g in total:
            total[g] += greeks[g] * pos["size"]
    
    return total

def delta_hedge_simulation(S_history, positions, rebalance_threshold=0.05) -> dict:
    """
    Delta 对冲模拟
    
    当 delta 偏移超过阈值时进行再平衡
    rebalance_threshold: 对冲阈值（5% 表示 delta 偏移超过 5% 时对冲）
    """
    rebalances = []
    pnl = 0
    hedge_costs = 0
    
    current_delta = 0
    
    for i, S in enumerate(S_history):
        # 重新计算当前 delta
        new_delta = calculate_portfolio_delta(positions, S)
        
        # 检查是否需要对冲
        if abs(new_delta - current_delta) > rebalance_threshold:
            # 执行对冲
            hedge_quantity = -(new_delta - current_delta)
            hedge_cost = hedge_quantity * S * 0.001  # 假设 0.1% 交易费
            hedge_costs += hedge_cost
            
            rebalances.append({
                "step": i,
                "price": S,
                "hedge_quantity": hedge_quantity,
                "hedge_cost": hedge_cost
            })
            
            current_delta = new_delta
        
        # 累计收益
        if i > 0:
            pnl += current_delta * (S - S_history[i-1])
    
    return {
        "total_rebalances": len(rebalances),
        "total_hedge_cost": hedge_costs,
        "net_pnl": pnl - hedge_costs,
        "rebalances": rebalances
    }
```

```python
def detect_skew_anomaly(smile_data: dict, spot_price: float) -> dict:
    """
    检测波动率偏斜异常
    
    正常 Skew：
    - OTM Put 的 IV > OTM Call 的 IV（下行风险溢价）
    - 深度 OTM Put 的 IV 显著高于 ATM
    
    异常信号：
    - Skew 变陡 → 市场担忧加剧
    - Skew 变平 → 市场乐观
    """
    strikes = np.array(smile_data["strikes"])
    iv_bid = np.array(smile_data["iv_bid"])
    
    # 分类 OTM Put 和 OTM Call
    atm_strike = spot_price
    otm_puts = strikes[strikes < spot_price]
    otm_calls = strikes[strikes > spot_price]
    
    if len(otm_puts) > 0:
        deep_otm_put_iv = iv_bid[strikes == otm_puts.min()][0]
    else:
        deep_otm_put_iv = 0
    
    atm_idx = np.argmin(np.abs(strikes - atm_strike))
    atm_iv = iv_bid[atm_idx] if atm_idx < len(iv_bid) else 0
    
    skew_spread = deep_otm_put_iv - atm_iv if atm_iv > 0 else 0
    
    return {
        "atm_iv_pct": atm_iv * 100,
        "deep_otm_put_iv_pct": deep_otm_put_iv * 100 if deep_otm_put_iv else None,
        "skew_spread_pct": skew_spread * 100,
        "signal": "skew_steep" if skew_spread > 0.1 else "skew_flat" if skew_spread < 0 else "normal",
        "interpretation": "市场担忧加剧，O Put 被大量买入" if skew_spread > 0.1 
                         else "市场乐观，看跌期权需求低"
    }

def term_structure_trade(iv_short: float, iv_long: float, horizon_days: int = 30) -> dict:
    """
    期限结构交易
    
    前端 IV > 后端 IV → Contango（正常）→ 卖近买远
    前端 IV < 后端 IV → Backwardation（倒挂）→ 买近卖远
    """
    structure = "contango" if iv_short > iv_long else "backwardation"
    spread = abs(iv_short - iv_long)
    
    # 交易逻辑
    if structure == "contango":
        trade = "卖近端 Call / 买远端 Call（Calendar Spread）"
        rationale = "Contango 结构下前端 IV 偏高，适合卖出"
    else:
        trade = "买近端 Call / 卖远端 Call（反向 Calendar Spread）"
        rationale = "Backwardation 结构通常伴随事件预期"
    
    return {
        "short_term_iv_pct": iv_short * 100,
        "long_term_iv_pct": iv_long * 100,
        "structure": structure,
        "spread_pct": spread * 100,
        "recommended_trade": trade,
        "rationale": rationale,
        "risk_per_trade": f"最大损失 = 权利金净差额（约 ${spread * horizon_days:.0f}/合约）"
    }
```

```python
def build_iron_condor(spot_price, expiry_days, target_iv=0.5, width_pct=0.05) -> dict:
    """
    构建铁鹰策略（Iron Condor）
    
    适合预期市场震荡、IV 中性或下降的环境
    
    盈亏图：
    - 标的价格在中间区间 → 盈利（卖方收益）
    - 标的价格大幅偏离 → 亏损
    """
    atm = spot_price
    width = spot_price * width_pct
    
    # 行权价
    put_lower = atm * (1 - width_pct * 2)  # Deep OTM Put
    put_upper = atm * (1 - width_pct)       # OTM Put
    call_lower = atm * (1 + width_pct)       # OTM Call
    call_upper = atm * (1 + width_pct * 2)  # Deep OTM Call
    
    # 定价
    put_lower_price = BlackScholes.put_price(spot_price, put_lower, expiry_days/365, 0.05, target_iv) * spot_price
    put_upper_price = BlackScholes.put_price(spot_price, put_upper, expiry_days/365, 0.05, target_iv) * spot_price
    call_lower_price = BlackScholes.call_price(spot_price, call_lower, expiry_days/365, 0.05, target_iv) * spot_price
    call_upper_price = BlackScholes.call_price(spot_price, call_upper, expiry_days/365, 0.05, target_iv) * spot_price
    
    # 净权利金 = 卖出 (put_upper + call_lower) - 买入 (put_lower + call_upper)
    net_premium = (put_upper_price + call_lower_price) - (put_lower_price + call_upper_price)
    
    # 最大盈利 = 净权利金
    max_profit = net_premium
    
    # 最大亏损 = (行权价间距 - 净权利金) * 合约乘数
    spread_width = width_pct * spot_price * 2
    max_loss = spread_width - net_premium
    
    return {
        "strategy": "Iron Condor",
        "legs": [
            {"type": "sell", "side": "put", "strike": put_upper, "premium": put_upper_price},
            {"type": "buy", "side": "put", "strike": put_lower, "premium": put_lower_price},
            {"type": "sell", "side": "call", "strike": call_lower, "premium": call_lower_price},
            {"type": "buy", "side": "call", "strike": call_upper, "premium": call_upper_price},
        ],
        "net_premium": net_premium,
        "max_profit": max_profit,
        "max_loss": max_loss,
        "profit_probability": 0.67,  # 约 2/3 概率盈利（基于标准差）
        "breakevens": [put_upper - net_premium, call_lower + net_premium],
        "best_when": "低波动环境、IV Rank 低（< 30%）、无重大事件"
    }

def build_short_straddle(spot_price, expiry_days, target_iv=0.5) -> dict:
    """
    构建 Short Straddle（卖跨式期权）
    预期：低波动，IV 高估
    """
    atm_put = BlackScholes.put_price(spot_price, spot_price, expiry_days/365, 0.05, target_iv) * spot_price
    atm_call = BlackScholes.call_price(spot_price, spot_price, expiry_days/365, 0.05, target_iv) * spot_price
    
    net_premium = atm_put + atm_call
    spread_width = spot_price  # ATM 行权价间距 = 标的价格
    
    return {
        "strategy": "Short Straddle",
        "legs": [
            {"type": "sell", "side": "put", "strike": spot_price, "premium": atm_put},
            {"type": "sell", "side": "call", "strike": spot_price, "premium": atm_call},
        ],
        "net_premium": net_premium,
        "max_profit": net_premium,
        "max_loss": float('inf'),  # 理论上无上限
        "breakevens": [spot_price - net_premium, spot_price + net_premium],
        "best_when": "IV > HV、预期低波动、期限短",
        "warning": "需严格止损，或在价格大幅波动时转为 Iron Condor"
    }

def build_ratio_spread(spot_price, expiry_days, target_iv=0.5, ratio=2) -> dict:
    """
    构建 Ratio Spread（比例价差）
    卖出 N 份期权，买入 1 份期权（或其他比例）
    
    例如 2:1 Call Ratio Spread
    - 买入 1 份 ATM Call
    - 卖出 2 份 OTM Call
    """
    otm_strike = spot_price * 1.05
    
    otm_call_price = BlackScholes.call_price(spot_price, otm_strike, expiry_days/365, 0.05, target_iv) * spot_price
    atm_call_price = BlackScholes.call_price(spot_price, spot_price, expiry_days/365, 0.05, target_iv) * spot_price
    
    net_premium = otm_call_price * 2 - atm_call_price
    
    return {
        "strategy": f"Call Ratio Spread ({ratio}:1)",
        "legs": [
            {"type": "buy", "side": "call", "strike": spot_price, "qty": 1, "premium": atm_call_price},
            {"type": "sell", "side": "call", "strike": otm_strike, "qty": ratio, "premium": otm_call_price * ratio},
        ],
        "net_premium": net_premium,
        "max_profit": "理论上无上限" if ratio > 1 else net_premium,
        "max_loss": abs(net_premium) if net_premium < 0 else (otm_strike - spot_price) - net_premium,
        "best_when": "温和上涨预期 + IV 适中"
    }
```

```python
def volatility_mean_reversion_strategy(
    current_iv: float,
    hv_30d: float,
    iv_history: list,
    iv_rank: float,
    direction: str = "sell"
) -> dict:
    """
    波动率均值回归策略
    
    逻辑：
    - IV Rank > 70%（IV 处于历史高位）→ 卖出期权（做空波动率）
    - IV Rank < 30%（IV 处于历史低位）→ 买入期权（做多波动率）
    
    参数：
    - current_iv: 当前隐含波动率
    - hv_30d: 30天历史波动率
    - iv_history: IV 历史序列（至少 252 天）
    - direction: "sell" 卖 IV / "buy" 买 IV
    """
    iv_mean = np.mean(iv_history)
    iv_std = np.std(iv_history)
    z_score = (current_iv - iv_mean) / iv_std
    
    # IV/HV 比率
    iv_hv_ratio = current_iv / hv_30d if hv_30d > 0 else 1
    
    # 策略信号
    signals = []
    
    if iv_rank > 70:
        signals.append("IV 处于历史高位 → 卖方机会")
    elif iv_rank < 30:
        signals.append("IV 处于历史低位 → 买方机会")
    else:
        signals.append("IV 中性 → 观望")
    
    if z_score > 1.5:
        signals.append(f"IV Z-Score = {z_score:.2f}（严重高估）")
    elif z_score < -1.5:
        signals.append(f"IV Z-Score = {z_score:.2f}（严重低估）")
    
    if iv_hv_ratio > 1.3:
        signals.append(f"IV/HV = {iv_hv_ratio:.2f}（IV 相对 HV 高估 > 30%）")
    elif iv_hv_ratio < 0.7:
        signals.append(f"IV/HV = {iv_hv_ratio:.2f}（IV 相对 HV 低估 > 30%）")
    
    # 策略选择
    if direction == "sell":
        if iv_rank > 60 and iv_hv_ratio > 1.1:
            recommended_strategy = "卖出 ATM Straddle"
            expected_return = iv_mean / current_iv - 1
        elif iv_rank > 50:
            recommended_strategy = "卖出 OTM Iron Condor"
            expected_return = 0.15
        else:
            recommended_strategy = "IV 偏低，不建议卖方"
            expected_return = 0
    else:
        if iv_rank < 40 and iv_hv_ratio < 0.9:
            recommended_strategy = "买入 ATM Straddle"
            expected_return = current_iv / iv_mean - 1
        elif iv_rank < 50:
            recommended_strategy = "买入 OTM Strangle"
            expected_return = 0.3
        else:
            recommended_strategy = "IV 偏高，不建议买方"
            expected_return = 0
    
    return {
        "current_iv_pct": current_iv * 100,
        "hv_30d_pct": hv_30d * 100,
        "iv_mean_pct": iv_mean * 100,
        "iv_rank_pct": iv_rank,
        "z_score": z_score,
        "iv_hv_ratio": iv_hv_ratio,
        "signals": signals,
        "recommended_strategy": recommended_strategy,
        "expected_return_pct": expected_return * 100,
        "risk_warnings": [
            "黑天鹅事件（波动率暴涨）可能导致巨大亏损",
            "需设置止损（如 IV 继续上涨 20% 强制平仓）",
            "考虑使用期权价差而非裸期权控制风险"
        ]
    }
```

```python
def calendar_spread_strategy(
    S, K, T_short, T_long, r, sigma
) -> dict:
    """
    日历价差（Calendar Spread）
    
    买入远端期权，卖出近端期权
    预期：近端期权到期时，标的在行权价附近
    """
    # 近端期权价格（短期）
    short_call = BlackScholes.call_price(S, K, T_short, r, sigma)
    short_put = BlackScholes.put_price(S, K, T_short, r, sigma)
    
    # 远端期权价格（长期）
    long_call = BlackScholes.call_price(S, K, T_long, r, sigma)
    long_put = BlackScholes.put_price(S, K, T_long, r, sigma)
    
    # 净成本
    net_cost_call = long_call - short_call
    net_cost_put = long_put - short_put
    
    # 最大盈利（标的 = 行权价时）
    if T_short > 0:
        intrinsic_short = max(S - K, 0) if short_call else max(K - S, 0)
        time_value_short = short_call - intrinsic_short if short_call else short_put
        # 近端到期时，远端剩余时间 = T_long - T_short
        T_remaining = T_long - T_short
        # 远端近端价格 = OTM 期权的剩余时间价值
        max_profit_call = BlackScholes.call_price(S, K, T_remaining, r, sigma) - net_cost_call
    else:
        max_profit_call = 0
    
    return {
        "calendar_call": {
            "short_premium": short_call,
            "long_premium": long_call,
            "net_cost": net_cost_call,
            "max_profit": max_profit_call,
            "max_loss": net_cost_call,
            "breakeven": K + net_cost_call,
            "best_when": "预期短期低波动，长期中性"
        }
    }
```

## 2. 期权基础概念

### 2.1 Call/Put
```
Call Option（看涨期权）：
  权利：到期日以约定价格买入标的
  买方：支付权利金（Premium），赚取价格上涨
  卖方：收取权利金，承担无限风险

Put Option（看跌期权）：
  权利：到期日以约定价格卖出标的
  买方：支付权利金，赚取价格下跌
  卖方：收取权利金，最大损失 = 行权价 - 权利金
```

### 2.2 关键术语
| 术语 | 解释 | 示例 |
|------|------|------|
| **Strike Price（行权价）** | 约定买卖价格 | $100,000 |
| **Expiration（到期日）** | 期权失效日期 | 2024-12-27 |
| **Premium（权利金）** | 买方支付给卖方的价格 | $2,500 / BTC |
| **ITM（价内）** | Call: 标的价格 > 行权价 | BTC $105,000, Strike $100,000 → ITM |
| **OTM（价外）** | Call: 标的价格 < 行权价 | BTC $95,000, Strike $100,000 → OTM |
| **ATM（平价）** | 标的价格 ≈ 行权价 | BTC $100,000, Strike $100,000 → ATM |

### 2.3 收益曲线（Payoff Diagram）
```
Call Buyer（买入看涨）：
  最大损失：权利金（有限）
  最大收益：无限（标的价格 ↑↑）
  
  收益 = Max(0, 标的价格 - 行权价) - 权利金

Put Buyer（买入看跌）：
  最大损失：权利金（有限）
  最大收益：行权价 - 权利金（标的价格 → 0）
  
  收益 = Max(0, 行权价 - 标的价格) - 权利金
```

## 3. Greeks 详解

### 3.1 Delta（Δ）- 价格敏感性
```
定义：标的价格变化 $1 → 期权价格变化多少
  
Call Delta: 0 到 1
Put Delta: -1 到 0
  
示例：
  BTC $100,000, Call Delta = 0.5
  → BTC 涨到 $101,000 → Call 价格增加 $500
```

### 3.2 Gamma（Γ）- Delta 变化率
```
定义：标的价格变化 $1 → Delta 变化多少
  
Gamma 最高：ATM（平价）期权
Gamma 最低：深度 ITM 或深度 OTM
  
意义：接近到期时，Gamma 激增 → 风险激增
```

### 3.3 Theta（Θ）- 时间衰减
```
定义：每过一天，期权价值损失多少
  
Theta 最高：ATM 期权
卖方最爱：卖出 ATM 期权，赚取 Theta
买方噩梦：持仓每天亏 Theta
```

### 3.4 Vega（ν）- 波动率敏感性
```
定义：隐含波动率（IV）变化 1% → 期权价格变化多少
  
Vega 最高：长期期权
  
策略：
  IV 低 → 买入期权（做多 Vega）
  IV 高 → 卖出期权（做空 Vega）
```

### 3.5 Rho（ρ）- 利率敏感性
```
定义：无风险利率变化 1% → 期权价格变化多少
  
影响较小（Web3 期权通常忽略）
```

### Python 实现：计算 Greeks
```python
# greeks.py
import numpy as np
from scipy.stats import norm

def black_scholes_greeks(S, K, T, r, sigma, option_type='call'):
    """
    Black-Scholes 模型计算 Greeks
    
    参数：
      S: 标的价格（Spot）
      K: 行权价（Strike）
      T: 剩余时间（年）
      r: 无风险利率
      sigma: 波动率（年化）
      option_type: 'call' 或 'put'
    
    返回：
      {'delta', 'gamma', 'theta', 'vega', 'rho'}
    """
    if T <= 0:
        # 已到期
        if option_type == 'call':
            delta = 1.0 if S > K else 0.0
        else:
            delta = -1.0 if S < K else 0.0
        return {'delta': delta, 'gamma': 0.0, 'theta': 0.0, 'vega': 0.0, 'rho': 0.0}
    
    
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    if option_type == 'call':
        delta = norm.cdf(d1)
        rho_sign = 1.0
    else:  # put
        delta = norm.cdf(d1) - 1
        rho_sign = -1.0
    
    
    gamma = norm.pdf(d1) / (S * sigma * np.sqrt(T))
    theta = (-S * norm.pdf(d1) * sigma / (2 * np.sqrt(T))
             - r * K * np.exp(-r * T) * norm.cdf(d2 if option_type == 'call' else -d2)) / 365
    vega = S * norm.pdf(d1) * np.sqrt(T) / 100  # 除以 100：IV 变化 1% → 价格变化
    rho = K * T * np.exp(-r * T) * norm.cdf(d2 if option_type == 'call' else -d2) * rho_sign / 100
    
    
    return {
        'delta': delta,
        'gamma': gamma,
        'theta': theta,
        'vega': vega,
        'rho': rho,
    }

# 示例
if __name__ == '__main__':
    S = 100000  # BTC 价格
    K = 105000  # 行权价
    T = 30 / 365  # 30 天
    r = 0.05  # 无风险利率 5%
    sigma = 0.6  # 波动率 60%（年化）
    
    
    greeks = black_scholes_greeks(S, K, T, r, sigma, option_type='call')
    
    
    print(f"Delta: {greeks['delta']:.4f}")
    print(f"Gamma: {greeks['gamma']:.6f}")

    print(f"Vega: {greeks['vega']:.4f}")
    print(f"Rho: {greeks['rho']:.4f}")
```

## 4. 期权策略类型学

### 策略 1：Covered Call（备兑看涨）
```
操作：
  持有 1 BTC
  + 卖出 1 BTC Call（收取权利金）
  
收益：
  标的价格 ↑ → 被行权，以 Strike 价格卖出 BTC（赚权利金 + 价格上涨到 Strike）
  标的价格 ↓ → 亏损（BTC 下跌），但权利金抵消部分亏损
  
适用：
  预期：价格横盘或小幅上涨
  目标：增收（收取权利金）
```

### 策略 2：Protective Put（保护性看跌）
```
操作：
  持有 1 BTC
  + 买入 1 BTC Put（支付权利金）
  
收益：
  标的价格 ↓ → Put 盈利（对冲 BTC 下跌）
  标的价格 ↑ → BTC 盈利，但损失权利金
  
适用：
  预期：不确定，但担心下跌
  目标：保险（类似买保险）
```

### 策略 3：Straddle（跨式）
```
操作：
  买入 1 Call（相同 Strike/到期日）
  + 买入 1 Put（相同 Strike/到期日）
  
收益：
  标的价格大幅波动（↑ 或 ↓）→ 盈利
  标的价格不动 → 亏损（损失双份权利金）
  
适用：
  预期：大波动，但方向不确定（如 CPI 数据发布前）
  目标：做多波动率
```

### 策略 4：Strangle（宽跨式）
```
操作：
  买入 1 OTM Call（Strike > 现价）
  + 买入 1 OTM Put（Strike < 现价）
  
与 Straddle 区别：
  成本更低（OTM 权利金便宜）
  但需要更大波动才能盈利
  
适用：
  预期：大波动，但不想付昂贵权利金
```

### 策略 5：Iron Condor（铁鹰）
```
操作：
  卖出 1 OTM Call
  + 买入 1 更 OTM Call（对冲风险）
  + 卖出 1 OTM Put
  + 买入 1 更 OTM Put（对冲风险）
  
收益：
  标的价格在窄幅区间 → 盈利（赚取双份权利金）
  标的价格突破区间 → 亏损（但有买入期权对冲，损失有限）
  
适用：
  预期：横盘（低波动）
  目标：赚取 Theta（时间衰减）
```

### 策略 6：Cash-Secured Put（现金担保 Put）
```
操作：
  卖出 1 Put
  + 在账户中存足够现金（以备被行权时买入 BTC）
  
收益：
  标的价格 > Strike → Put 过期，赚取权利金
  标的价格 < Strike → 被行权，以 Strike 价格买入 BTC（实际成本 = Strike - 权利金）
  
适用：
  预期：价格不会跌破 Strike
  目标：以理想价格买入 BTC（同时赚取权利金）
```

## 5. 波动率交易（Volatility Trading）

### 5.1 IV vs HV（隐含波动率 vs 历史波动率）
```
IV（Implied Volatility）：
  市场预期的未来波动率
  来源：从期权价格反推（B-S 公式反解）

HV（Historical Volatility）：
  过去实际波动率
  计算：过去 N 天收益率的标准差（年化）
```

### 5.2 VRP（Volatility Risk Premium）
```
VRP = IV - HV
  
含义：
  VRP > 0 → 市场预期波动 > 实际波动 → 卖出期权（做空波动率）
  VRP < 0 → 市场预期波动 < 实际波动 → 买入期权（做多波动率）
  
策略：
  VRP > 0.2 → 卖出 Straddle（做空波动率）
  VRP < -0.1 → 买入 Straddle（做多波动率）
```

### 5.3 代码模板：VRP 策略
```python
# vrp_strategy.py
import requests
import numpy as np

def fetch_deribit_iv(symbol, strike, expiration):
    """
    获取 Deribit 隐含波动率（IV）
    """
    url = "https://test.deribit.com/api/v2/public/get_order_book"
    params = {
        'instrument_name': f"{symbol}-{expiration}-{strike}-C",
        'depth': 1,
    }
    resp = requests.get(url, params=params, timeout=5)
    data = resp.json()['result']
    
    iv = data['mark_iv']  # 隐含波动率（年化）
    return iv

def calculate_hv(candles, lookback=30):
    """
    计算历史波动率（HV）
    """
    closes = np.array([c['close'] for c in candles[-lookback:]])
    returns = np.diff(np.log(closes))
    hv = np.std(returns) * np.sqrt(365)  # 年化
    return hv

def vrp_strategy(symbol, candles, threshold=0.2):
    """
    VRP 策略（波动率风险溢价）
    
    参数：
      threshold: VRP 阈值（默认 0.2 = 20%）
    
    返回：
      signal: 'SELL_VOL'（做空波动率）/ 'BUY_VOL'（做多波动率）/ None
    """
    # 获取 IV（简化：假设已获取）
    iv = fetch_deribit_iv(symbol, strike=100000, expiration='27DEC24')
    
    # 计算 HV
    hv = calculate_hv(candles, lookback=30)
    
    # 计算 VRP
    vrp = iv - hv
    
    if vrp > threshold:
        signal = 'SELL_VOL'  # 卖出期权（做空波动率）
    elif vrp < -0.1:
        signal = 'BUY_VOL'   # 买入期权（做多波动率）
    else:
        signal = None
    
    return {
        'signal': signal,
        'iv': iv,
        'hv': hv,
        'vrp': vrp,
    }

# 使用
if __name__ == '__main__':
    candles = fetch_klines('BTCUSDT', interval='1d', limit=90)
    result = vrp_strategy('BTC', candles, threshold=0.2)
    
    print(f"IV: {result['iv']*100:.1f}%")
    print(f"HV: {result['hv']*100:.1f}%")
    print(f"VRP: {result['vrp']*100:.1f}%")
    print(f"信号: {result['signal']}")
```

### 模板：Covered Call 回测
```python
# covered_call_backtest.py
import numpy as np
from datetime import datetime

class CoveredCallBacktest:
    def __init__(self, btc_candles, option_data, strike, expiration_days):
        """
        Covered Call 回测
        
        参数：
          btc_candles: BTC K 线
          option_data: 期权价格数据（权利金）
          strike: 行权价
          expiration_days: 距离到期日天数
        """
        self.btc_candles = btc_candles
        self.option_data = option_data
        self.strike = strike
        self.expiration_days = expiration_days
        
        self.initial_btc = 1.0  # 初始 1 BTC
        self.cash = 0.0  # 现金（USD）
        self.premium_collected = 0.0  # 收取的权利金
    
    def run(self):
        """运行回测"""
        results = []
        
        for i, candle in enumerate(self.btc_candles):
            btc_price = candle['close']
            
            # 卖出 Call，收取权利金（简化：假设在第 0 天卖出）
            if i == 0:
                premium = self.option_data[i]['premium']
                self.premium_collected += premium
                self.cash += premium * self.initial_btc
            
            
            # 检查是否被行权
            if i >= self.expiration_days:
                if btc_price > self.strike:
                    # 被行权：以 Strike 价格卖出 BTC
                    self.cash += self.initial_btc * self.strike
                    self.initial_btc = 0.0
                    
                    results.append({
                        'day': i,
                        'btc_price': btc_price,
                        'btc_amount': self.initial_btc,
                        'cash': self.cash,
                        'total_value': self.cash,  # BTC 已被行权卖出
                    })
                    break
            
            
            # 未被行权：继续持有 BTC
            total_value = self.cash + self.initial_btc * btc_price
            results.append({
                'day': i,
                'btc_price': btc_price,
                'btc_amount': self.initial_btc,
                'cash': self.cash,
                'total_value': total_value,
            })
        
        
        return results
    
    def calculate_return(self, results):
        """计算收益"""
        initial_value = self.btc_candles[0]['close']  # 初始 1 BTC 价值
        final_value = results[-1]['total_value']
        
        total_return = (final_value - initial_value) / initial_value * 100
        annualized_return = total_return / (len(results) / 365)
        
        return {
            'total_return_pct': total_return,
            'annualized_return_pct': annualized_return,
            'final_value': final_value,
        }

# 使用
if __name__ == '__main__':
    # 加载数据（简化）
    btc_candles = fetch_klines('BTCUSDT', interval='1d', limit=90)
    option_data = [{'premium': 0.05}] * 90  # 假设权利金 0.05 BTC
    
    backtest = CoveredCallBacktest(
        btc_candles=btc_candles,
        option_data=option_data,
        strike=105000,  # 行权价 $105,000
        expiration_days=30,  # 30 天后到期
    )
    
    results = backtest.run()
    returns = backtest.calculate_return(results)
    
    print(f"总收益: {returns['total_return_pct']:.2f}%")
    print(f"年化收益: {returns['annualized_return_pct']:.2f}%")
    print(f"最终价值: ${returns['final_value']:,.2f}")
```

## 8. 期权风险管理

### 8.1 风险指标
| 风险 | 说明 | 管理方法 |
|------|------|----------|
| **Delta Risk** | 标的价格变化风险 | Delta Hedge（用期货对冲） |
| **Gamma Risk** | Delta 变化加速风险 | 避免临近到期（Gamma 激增） |
| **Theta Risk** | 时间衰减风险（买方） | 避免买入临近到期期权 |
| **Vega Risk** | 波动率变化风险 | 做多/做空 Vega 要控制仓位 |
| **Pin Risk** | 到期时价格接近 Strike | 提前平仓或滚动到下一期 |

### 8.2 仓位管理
```
单笔期权仓位 ≤ 总资金 5%
期权总仓位 ≤ 总资金 20%
```

### 8.3 止损
```
期权买方：
  损失超过权利金 50% → 止损
  
期权卖方：
  浮亏超过收取权利金的 2 倍 → 止损
```

## 9. 总结与最佳实践

### 期权策略选择速查表
| 市场预期 | 推荐策略 | 风险 |
|---------|---------|------|
| **横盘（低波动）** | Covered Call / Iron Condor | 低 |
| **小幅上涨** | Covered Call | 低 |
| **大幅波动（方向不明）** | Straddle / Strangle | 中 |
| **波动率过高** | 卖出 Straddle（做空波动率） | 高 |
| **波动率过低** | 买入 Straddle（做多波动率） | 中 |
| **担心下跌但想持有** | Protective Put | 低（保险成本） |

### 最佳实践
1. **先模拟盘**：期权复杂度高，先模拟再实盘
2. **控制仓位**：单笔 ≤ 5% 总资金
3. **理解 Greeks**：Delta/Gamma/Theta/Vega 必须懂
4. **避免临近到期**：Gamma 风险激增
5. **关注 IV**：IV 高 → 卖出；IV 低 → 买入

### 常见错误
```
❌ 买入临近到期期权（Theta 衰减太快）
❌ 卖出未对冲的 Naked Call（风险无限）
❌ 忽略 Vega 风险（IV 突然下降 → 亏损）
❌ 重仓单一期权（黑天鹅风险）
```

### 8.3 Greeks 详解

| Greek | 含义 | 对冲方法 |
|--------|------|----------|
| **Delta (Δ)** | 标的涨 $1，期权涨多少 | 卖出现货/期货对冲 |
| **Gamma (Γ)** | Delta 变化速度 | Gamma  scalping（高 Gamma 时频繁对冲） |
| **Vega (ν)** | 波动率涨 1%，期权涨多少 | 交易波动率（做多/做空 IV） |
| **Theta (θ)** | 每天时间衰减 | 卖出期权（收 Theta） |
| **Rho (ρ)** | 利率涨 1%，期权涨多少 | 利率对冲（很少用） |

**Delta 中性示例**：
```
持有：10 张 BTC Call（Delta = 0.68）
总 Delta = 10 × 0.68 = 6.8

对冲：卖出 6.8 BTC 现货
新的 Delta = 6.8 - 6.8 = 0 ✅（中性）
```

---

### 8.4 IV Rank 交易系统

**IV Rank 计算**：
```
IV Rank = (当前 IV - 52周最低 IV) / (52周最高 IV - 52周最低 IV) × 100
```

| IV Rank | 含义 | 操作建议 |
|---------|------|----------|
| > 80 | 波动率极高 | 卖出期权（做空波动率） |
| 50-80 | 波动率偏高 | 可考虑卖出 |
| 20-50 | 波动率正常 | 不操作或做方向 |
| < 20 | 波动率极低 | 买入期权（做多波动率） |

**实战示例**：
```python

    print("建议：买入 Straddle（做多波动率）")
```

```python
twap_plan = hedger.calc_twap_plan(
    total_quantity=2.35,
    duration_minutes=30,
    interval_minutes=5
)
for step in twap_plan:
    print(f"T+{step['time']}：卖出 {step['quantity']} BTC")

# 4. IV Rank 分析
analyzer = IVRankAnalyzer()
iv_rank = analyzer.calculate_iv_rank('BTC', days=365)
print(f"IV Rank：{iv_rank:.0f}")
if iv_rank > 80:
    print("建议：卖出波动率")
```

### 8.6 对冲执行最佳实践

1. **阈值设置**：
   - 保守：threshold = 0.03（3%）
   - 标准：threshold = 0.05（5%）
   - 激进：threshold = 0.10（10%）

2. **对冲频率**：
   - 高 Gamma 期权：每日对冲
   - 低 Gamma 期权：每周对冲
   - 临近到期：每日多次对冲

3. **冲击成本控制**：
   - 单笔 > 10 BTC → 使用 TWAP
   - 单笔 > 50 BTC → 使用 Iceberg 单

4. **极端行情处理**：
   - 波动率暴涨（IV Rank > 90）→ 加快对冲频率
   - 流动性枯竭 → 暂停对冲，等待恢复
