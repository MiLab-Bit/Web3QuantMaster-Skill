```python
class StablecoinArbitrage:
    """稳定币跨链/跨交易所套利"""
    
    def __init__(self):
        self.chain_rpc = {
            "Ethereum": "https://eth.llamarpc.com",
            "Arbitrum": "https://arb1.arbitrum.io/rpc",
            "Optimism": "https://mainnet.optimism.io",
            "Polygon": "https://polygon-rpc.com",
            "Solana": "https://api.mainnet-beta.solana.com"
        }
    
    def get_cross_chain_swap_rate(self, token: str, from_chain: str, to_chain: str) -> dict:
        """
        获取跨链兑换汇率
        
        考虑成本：
        1. 跨链桥费用
        2. Gas 费
        3. 滑点
        """
        bridges = {
            ("Ethereum", "Arbitrum"): {
                "bridges": ["Arbitrum Bridge", "Hop Exchange", "Across Protocol"],
                "avg_time_minutes": 10,
                "avg_fee_usd": 5
            },
            ("Ethereum", "Polygon"): {
                "bridges": ["Polygon Bridge", "Hop Exchange"],
                "avg_time_minutes": 7,
                "avg_fee_usd": 1
            },
            ("Ethereum", "Solana"): {
                "bridges": ["Wormhole"],
                "avg_time_minutes": 20,
                "avg_fee_usd": 3
            }
        }
        
        # 获取 DEX 汇率
        from_dex_rate = get_dex_price(token, from_chain)
        to_dex_rate = get_dex_price(token, to_chain)
        
        bridge_info = bridges.get((from_chain, to_chain), {})
        
        # 净收益计算
        gross_spread = to_dex_rate - from_dex_rate
        net_spread = gross_spread - bridge_info.get("avg_fee_usd", 10)
        
        return {
            "from_chain": from_chain,
            "to_chain": to_chain,
            "from_price": from_dex_rate,
            "to_price": to_dex_rate,
            "gross_spread_usd": gross_spread,
            "bridge_fee_usd": bridge_info.get("avg_fee_usd", 0),
            "net_spread_usd": net_spread,
            "avg_time_minutes": bridge_info.get("avg_time_minutes", 0),
            "viable": net_spread > 0 and bridge_info.get("avg_time_minutes", 0) < 60
        }
    
    def detect_peg_deviation(self, stablecoin: str) -> list:
        """
        检测稳定币脱锚偏差
        
        检查多个 DEX 的价格是否偏离 $1.00
        """
        # 从 Uniswap、Curve、SushiSwap 等获取价格
        prices = {}
        
        for dex in ["uniswap_v3", "curve", "sushiswap"]:
            try:
                price = get_stablecoin_price(dex, stablecoin)
                prices[dex] = price
            except:
                pass
        
        if not prices:
            return []
        
        avg_price = np.mean(list(prices.values()))
        deviations = {dex: (price - 1.0) * 100 for dex, price in prices.items()}
        
        # 最大偏差
        max_deviation = max(abs(d) for d in deviations.values())
        most_deviated_dex = max(deviations.items(), key=lambda x: abs(x[1]))[0]
        
        return {
            "stablecoin": stablecoin,
            "prices": prices,
            "avg_price_usd": avg_price,
            "max_deviation_pct": max_deviation,
            "most_deviated_dex": most_deviated_dex,
            "arbitrage_opportunity": max_deviation > 0.1,  # > 0.1% 偏差时存在套利机会
            "action": "买低卖高" if deviations.get(most_deviated_dex, 0) < 0 else "观望"
        }
    
    def execute_peg_arbitrage(self, stablecoin: str, capital=10000) -> dict:
        """
        执行脱锚套利
        
        策略：
        - 如果 DEX 价格 < $1 → 买入（预期回归 $1）
        - 如果 DEX 价格 > $1 → 卖出（做空）
        
        ⚠️ 警告：算法稳定币可能不会回归！
        """
        deviation = self.detect_peg_deviation(stablecoin)
        
        if not deviation["arbitrage_opportunity"]:
            return {"action": "NO_TRADE", "reason": "无显著偏差"}
        
        # 获取该稳定币类型
        is_algorithmic = stablecoin in ["FRAX", "USDD", "USDE"]
        
        if is_algorithmic:
            # 算法稳定币：脱锚可能是趋势性的
            return {
                "action": "CAUTION",
                "reason": f"{stablecoin} 是算法稳定币，脱锚可能是系统性风险",
                "historical_deviation_days": check_historical_deviation_days(stablecoin),
                "recommendation": "仅在偏差 > 2% 且有强力催化剂时入场"
            }
        
        # 法币/加密抵押稳定币
        deviation_pct = deviation["max_deviation_pct"]
        position_size = capital * min(deviation_pct / 0.01, 1.0)  # 偏差越大，仓位越大
        
        return {
            "action": "EXECUTE",
            "stablecoin": stablecoin,
            "direction": "BUY" if deviation["most_deviated_dex"] else "SELL",
            "dex": deviation["most_deviated_dex"],
            "entry_price": deviation["avg_price_usd"],
            "target_price": 1.0,
            "position_size": position_size,
            "expected_return_pct": deviation_pct * 100,
            "stop_loss": 0.995 if deviation_pct > 0 else 1.005,
            "risk_warnings": [
                "滑点风险（大量交易会进一步偏离价格）",
                "Gas 费消耗（尤其 ETH 主网）",
                "需等待价格回归（时间不确定）"
            ]
        }
```

### 稳定币三角套利

```python
def detect_triangular_arb_opportunity(pairs: dict) -> dict:
    """
    检测三角套利机会
    
    例如：USDT → USDC → DAI → USDT
    如果三者之间的汇率乘积不等于 1，则存在套利机会
    
    例：
    USDT/USDC = 0.999
    USDC/DAI = 0.999
    DAI/USDT = 0.999
    乘积 = 0.999^3 = 0.997 → 约 0.3% 套利空间
    """
    # 获取汇率
    usdt_usdc = get_exchange_rate("USDT", "USDC")  # USDT → USDC
    usdc_dai = get_exchange_rate("USDC", "DAI")    # USDC → DAI
    dai_usdt = get_exchange_rate("DAI", "USDT")    # DAI → USDT
    
    # 计算乘积
    product = usdt_usdc * usdc_dai * dai_usdt
    
    # 费用扣除
    trading_fee = 0.003  # 0.3% per trade (Uniswap V2)
    total_fees = 3 * trading_fee
    net_product = product - total_fees
    
    return {
        "rates": {
            "USDT/USDC": usdt_usdc,
            "USDC/DAI": usdc_dai,
            "DAI/USDT": dai_usdt
        },
        "gross_product": product,
        "total_fees_pct": total_fees * 100,
        "net_product": net_product,
        "arb_profit_pct": (net_product - 1) * 100,
```

## 七、稳定币夏普比率策略

### 稳定币收益轮动

```python
def stablecoin_rotation_strategy(
    capital: float = 100000,
    rebalance_frequency: str = "weekly",
    min_yield_threshold: float = 0.03,
    risk_adjusted: bool = True
) -> dict:
    """
    稳定币收益轮动策略
    
    逻辑：
    1. 每周扫描各协议稳定币存款利率
    2. 排除高风险协议（未审计、新协议 TVL 低）
    3. 分配到风险调整后收益最高的协议
    4. 设置再平衡阈值（如：收益率差距 > 2% 时触发）
    
    参数：
    - capital: 总资金
    - rebalance_frequency: 再平衡频率（daily/weekly/monthly）
    - min_yield_threshold: 最低收益率阈值（过滤噪音）
    - risk_adjusted: 是否进行风险调整
    """
    yields = StablecoinYieldTracker().get_current_yields()
    best_yields = StablecoinYieldTracker().find_best_stablecoin_yield()
    
    # 按风险调整收益排序
    sorted_protocols = sorted(
        best_yields["best_yields"],
        key=lambda x: x.get("risk_adjusted_apy", x["raw_apy"]),
        reverse=True
    )
    
    # 取前3名
    top_protocols = sorted_protocols[:3]
    
    # 分配逻辑
    # 第1名：60%
    # 第2名：30%
    # 第3名：10%
    allocation_pct = [0.60, 0.30, 0.10]
    
    allocations = {}
    total_expected_apy = 0
    
    for i, protocol in enumerate(top_protocols[:3]):
        alloc_amount = capital * allocation_pct[i]
        expected_return = alloc_amount * protocol["risk_adjusted_apy"]
        
        allocations[f"#{i+1}: {protocol['protocol']} ({protocol['stablecoin']})"] = {
            "amount_usd": alloc_amount,
            "raw_apy": protocol["raw_apy"],
            "risk_adjusted_apy": protocol["risk_adjusted_apy"],
            "expected_annual_return": expected_return,
            "chain": protocol.get("chain", "Ethereum"),
            "risk_score": protocol.get("risk_score", 0.5)
        }
        
        total_expected_apy += expected_return
    
    # 计算组合整体指标
    overall_apy = total_expected_apy / capital
    
    return {
        "strategy": "稳定币收益轮动",
        "total_capital": capital,
        "rebalance_frequency": rebalance_frequency,
        "allocations": allocations,
        "expected_annual_return_usd": total_expected_apy,
        "expected_annual_apy_pct": overall_apy * 100,
        "rebalance_trigger": "当第一名收益率被第二名超过 2% 以上时",
        "key_risks": [
            "智能合约风险（优先选择已审计协议）",
            "TVL 突然下降（可能预示撤资潮）",
            "稳定币脱锚风险（优先选择 USDT/USDC/DAI）",
            "无常损失（如果提供 LP）"
        ],
        "best_practices": [
            "分散到 2-3 个协议",
            "优先选择 ETH 主网或 L2（有审计记录）",
            "避免新协议（< 3 个月 TVL > $10M）",
            "保持 10% 资金在主流 CEX 作为备份"
        ]
    }
```
