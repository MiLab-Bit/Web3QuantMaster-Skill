```python
# trend_strategy.py
import pandas as pd
import numpy as np
import json
from datetime import datetime, timedelta

class TrendFollowingStrategy:
    """
    双均线趋势跟踪策略
    
    策略逻辑：
    - 快速 MA 上穿慢速 MA → 做多
    - 快速 MA 下穿慢速 MA → 做空/平多
    """
    
    def __init__(self, fast_period: int = 10, slow_period: int = 30, 
                 initial_capital: float = 10000):
        self.fast_period = fast_period
        self.slow_period = slow_period
        self.initial_capital = initial_capital
        self.position = 0  # 0 = 空仓, 1 = 多头, -1 = 空头
        
        # 历史数据
        self.price_history = []
        self.trades = []
        self.equity_curve = []
    
    def calculate_ma(self, prices: list, period: int) -> float:
        """计算移动平均"""
        if len(prices) < period:
            return None
        return np.mean(prices[-period:])
    
    def signal(self, current_price: float) -> str:
        """
        生成交易信号
        
        返回：
        - "LONG" - 做多
        - "SHORT" - 做空
        - "CLOSE_LONG" - 平多
        - "CLOSE_SHORT" - 平空
        - "HOLD" - 观望
        """
        self.price_history.append(current_price)
        
        if len(self.price_history) < self.slow_period:
            return "HOLD"
        
        fast_ma = self.calculate_ma(self.price_history, self.fast_period)
        slow_ma = self.calculate_ma(self.price_history, self.slow_period)
        
        if fast_ma is None or slow_ma is None:
            return "HOLD"
        
        # 计算上一根K线的 MA
        prev_fast_ma = np.mean(self.price_history[-self.fast_period-1:-1])
        prev_slow_ma = np.mean(self.price_history[-self.slow_period-1:-1])
        
        # 交叉检测
        if prev_fast_ma <= prev_slow_ma and fast_ma > slow_ma:
            if self.position != 1:
                self.position = 1
                return "LONG"
        
        if prev_fast_ma >= prev_slow_ma and fast_ma < slow_ma:
            if self.position == 1:
                self.position = 0
                return "CLOSE_LONG"
        
        return "HOLD"
    
    def run_backtest(self, price_data: list, timestamps: list = None) -> dict:
        """
        运行回测
        
        price_data: 价格序列
        timestamps: 时间戳（可选）
        """
        capital = self.initial_capital
        position = 0
        entry_price = 0
        trades = []
        equity_curve = []
        
        for i, price in enumerate(price_data):
            signal = self.signal(price)
            
            if signal == "LONG" and position == 0:
                position = 1
                entry_price = price
                trades.append({
                    "type": "BUY",
                    "price": price,
                    "timestamp": timestamps[i] if timestamps else i,
                    "capital": capital
                })
            
            elif signal == "CLOSE_LONG" and position == 1:
                pnl = capital * (price - entry_price) / entry_price
                capital += pnl
                trades.append({
                    "type": "SELL",
                    "price": price,
                    "pnl": pnl,
                    "pnl_pct": pnl / (capital - pnl) * 100,
                    "timestamp": timestamps[i] if timestamps else i
                })
                position = 0
            
            equity_curve.append({
                "timestamp": timestamps[i] if timestamps else i,
                "price": price,
                "equity": capital,
                "position": position
            })
        
        # 计算绩效指标
        returns = np.diff([e["equity"] for e in equity_curve]) / [e["equity"] for e in equity_curve[:-1]]
        
        return {
            "final_capital": capital,
            "total_return_pct": (capital - self.initial_capital) / self.initial_capital * 100,
            "total_trades": len(trades),
            "winning_trades": len([t for t in trades if t.get("pnl", 0) > 0]),
            "losing_trades": len([t for t in trades if t.get("pnl", 0) < 0]),
            "win_rate": len([t for t in trades if t.get("pnl", 0) > 0]) / len(trades) * 100 if trades else 0,
            "max_drawdown_pct": self.calculate_max_drawdown(equity_curve),
            "sharpe_ratio": self.calculate_sharpe(returns),
            "trades": trades,
            "equity_curve": equity_curve
        }
    
    @staticmethod
    def calculate_max_drawdown(equity_curve: list) -> float:
        """计算最大回撤"""
        equity = [e["equity"] for e in equity_curve]
        peak = equity[0]
        max_dd = 0
        
        for e in equity:
            if e > peak:
                peak = e
            dd = (peak - e) / peak
            if dd > max_dd:
                max_dd = dd
        
        return max_dd * 100
    
    @staticmethod
    def calculate_sharpe(returns: list, risk_free_rate: float = 0.02) -> float:
        """计算夏普比率"""
        if len(returns) < 2:
            return 0
        
        returns = np.array(returns)
        excess_return = np.mean(returns) - risk_free_rate / 252
        sharpe = excess_return / np.std(returns) * np.sqrt(252) if np.std(returns) > 0 else 0
        
        return sharpe

    np.random.seed(42)

    print(f"胜率: {results['win_rate']:.1f}%")
    print(f"最大回撤: {results['max_drawdown_pct']:.2f}%")
    print(f"夏普比率: {results['sharpe_ratio']:.2f}")
```

```python
class BollingerBandStrategy:
    """
    布林带突破策略
    
    策略逻辑：
    - 价格突破上轨 → 做多（趋势延续）
    - 价格跌破下轨 → 做空（反弹）
    - 止损：入场价 ± 2 * ATR
    """
    
    def __init__(self, bb_period: int = 20, bb_std: float = 2.0,
                 atr_period: int = 14, atr_multiplier: float = 2.0,
                 initial_capital: float = 10000):
        self.bb_period = bb_period
        self.bb_std = bb_std
        self.atr_period = atr_period
        self.atr_multiplier = atr_multiplier
        self.initial_capital = initial_capital
        
        self.prices = []
        self.highs = []
        self.lows = []
        self.position = 0
        self.entry_price = 0
    
    def calculate_bollinger_bands(self) -> tuple:
        """计算布林带"""
        if len(self.prices) < self.bb_period:
            return None, None, None
        
        recent = self.prices[-self.bb_period:]
        sma = np.mean(recent)
        std = np.std(recent)
        
        upper = sma + self.bb_std * std
        lower = sma - self.bb_std * std
        
        return upper, sma, lower
    
    def calculate_atr(self) -> float:
        """计算 ATR"""
        if len(self.prices) < self.atr_period + 1:
            return None
        
        tr_list = []
        for i in range(len(self.prices) - self.atr_period, len(self.prices)):
            if i > 0:
                tr = max(
                    self.highs[i] - self.lows[i],
                    abs(self.highs[i] - self.prices[i-1]),
                    abs(self.lows[i] - self.prices[i-1])
                )
                tr_list.append(tr)
        
        return np.mean(tr_list)
    
    def signal(self, high: float, low: float, close: float) -> str:
        """
        生成交易信号
        """
        self.highs.append(high)
        self.lows.append(low)
        self.prices.append(close)
        
        if len(self.prices) < max(self.bb_period, self.atr_period) + 1:
            return "HOLD"
        
        upper, middle, lower = self.calculate_bollinger_bands()
        atr = self.calculate_atr()
        
        if upper is None or atr is None:
            return "HOLD"
        
        # 价格突破上轨
        if close > upper and self.position == 0:
            self.position = 1
            self.entry_price = close
            return "LONG"
        
        # 价格跌破下轨
        if close < lower and self.position == 0:
            self.position = -1
            self.entry_price = close
            return "SHORT"
        
        # 止损检查
        if self.position == 1 and close < self.entry_price - self.atr_multiplier * atr:
            self.position = 0
            return "STOP_LOSS_LONG"
        
        if self.position == -1 and close > self.entry_price + self.atr_multiplier * atr:
            self.position = 0
            return "STOP_LOSS_SHORT"
        
        return "HOLD"
```

```python
class MeanReversionStrategy:
    """
    布林带均值回归策略
    
    策略逻辑：
    - 价格触及下轨 → 买入（预期回归均线）
    - 价格触及上轨 → 卖出（预期回归均线）
    - 止损：下破下轨（续跌）或上穿上轨（续涨）
    """
    
    def __init__(self, period: int = 20, std_dev: float = 2.0,
                 initial_capital: float = 10000, stop_loss_pct: float = 0.02):
        self.period = period
        self.std_dev = std_dev
        self.initial_capital = initial_capital
        self.stop_loss_pct = stop_loss_pct
        
        self.prices = []
        self.position = 0
        self.entry_price = 0
    
    def calculate_bands(self) -> tuple:
        """计算布林带"""
        if len(self.prices) < self.period:
            return None, None, None
        
        recent = self.prices[-self.period:]
        sma = np.mean(recent)
        std = np.std(recent)
        
        return sma + self.std_dev * std, sma, sma - self.std_dev * std
    
    def signal(self, close_price: float) -> str:
        """生成信号"""
        self.prices.append(close_price)
        
        if len(self.prices) < self.period:
            return "HOLD"
        
        upper, middle, lower = self.calculate_bands()
        
        # 触及下轨 → 买入
        if close_price <= lower and self.position == 0:
            self.position = 1
            self.entry_price = close_price
            return "BUY"
        
        # 触及上轨 → 卖出
        if close_price >= upper and self.position == 0:
            self.position = -1
            self.entry_price = close_price
            return "SHORT"
        
        # 回归均线 → 平仓
        if self.position == 1 and close_price >= middle:
            self.position = 0
            return "CLOSE_LONG"
        
        if self.position == -1 and close_price <= middle:
            self.position = 0
            return "CLOSE_SHORT"
        
        # 止损
        if self.position == 1 and close_price < self.entry_price * (1 - self.stop_loss_pct):
            self.position = 0
            return "STOP_LOSS"
        
        if self.position == -1 and close_price > self.entry_price * (1 + self.stop_loss_pct):
            self.position = 0
            return "STOP_LOSS"
        
        return "HOLD"
    
    def run_backtest(self, prices: list) -> dict:
        """运行回测"""
        capital = self.initial_capital
        position = 0
        entry_price = 0
        trades = []
        
        for price in prices:
            signal = self.signal(price)
            
            if signal == "BUY":
                position = 1
                entry_price = price
                trades.append({"type": "BUY", "price": price})
            
            elif signal == "CLOSE_LONG":
                pnl = capital * (price - entry_price) / entry_price
                capital += pnl
                trades.append({"type": "SELL", "price": price, "pnl": pnl})
                position = 0
            
            elif signal == "STOP_LOSS":
                pnl = capital * (price - entry_price) / entry_price
                capital += pnl
                trades.append({"type": "STOP_LOSS", "price": price, "pnl": pnl})
                position = 0
        
        return {
            "final_capital": capital,
            "return_pct": (capital - self.initial_capital) / self.initial_capital * 100,
            "trades": trades,
            "win_rate": len([t for t in trades if t.get("pnl", 0) > 0]) / len(trades) * 100 if trades else 0
        }
```

```python
class RSIMeanReversionStrategy:
    """
    RSI 均值回归策略
    
    策略逻辑：
    - RSI < 30 → 超卖 → 买入
    - RSI > 70 → 超买 → 卖出
    - RSI < 20 → 极超卖 → 重仓买入
    - RSI > 80 → 极超买 → 重仓卖出
    """
    
    def __init__(self, period: int = 14, initial_capital: float = 10000):
        self.period = period
        self.initial_capital = initial_capital
        self.prices = []
        self.position = 0
        self.entry_price = 0
    
    def calculate_rsi(self) -> float:
        """计算 RSI"""
        if len(self.prices) < self.period + 1:
            return None
        
        prices = self.prices[-self.period-1:]
        deltas = np.diff(prices)
        gains = np.where(deltas > 0, deltas, 0)
        losses = np.where(deltas < 0, -deltas, 0)
        
        avg_gain = np.mean(gains)
        avg_loss = np.mean(losses)
        
        if avg_loss == 0:
            return 100
        
        rs = avg_gain / avg_loss
        return 100 - (100 / (1 + rs))
    
    def signal(self, close_price: float) -> str:
        """生成信号"""
        self.prices.append(close_price)
        
        rsi = self.calculate_rsi()
        if rsi is None:
            return "HOLD"
        
        # RSI 极低 → 买入
        if rsi < 30 and self.position == 0:
            self.position = 1
            self.entry_price = close_price
            return "BUY"
        
        # RSI 极低 → 重仓
        if rsi < 20 and self.position == 1:
            # 可以加仓
            return "ADD"
        
        # RSI 回归 → 平仓
        if self.position == 1 and rsi > 50:
            self.position = 0
            return "CLOSE"
        
        return "HOLD"
```

```python
class GridStrategy:
    """
    网格交易策略
    
    策略逻辑：
    1. 设置价格区间和网格数量
    2. 价格每下跌一格 → 买入
    3. 价格每上涨一格 → 卖出
    4. 无论方向，持续套利
    
    优点：震荡市场收益稳定
    缺点：单边趋势可能亏损严重
    """
    
    def __init__(self, lower_price: float, upper_price: float, grid_count: int = 10,
                 initial_capital: float = 10000, base_asset: str = "USDT"):
        self.lower_price = lower_price
        self.upper_price = upper_price
        self.grid_count = grid_count
        self.grid_spacing = (upper_price - lower_price) / grid_count
        self.initial_capital = initial_capital
        self.base_asset = base_asset
        
        # 网格状态
        self.grids = {}
        for i in range(1, grid_count + 1):
            grid_price = lower_price + i * self.grid_spacing
            self.grids[grid_price] = {"bought": False, "amount": 0}
        
        # 持仓
        self.base_balance = 0  # 基础资产（如 ETH）
        self.quote_balance = initial_capital  # 报价资产（如 USDT）
        
        # 记录
        self.trades = []
        self.profit_per_grid = []
    
    def get_current_grid(self, price: float) -> int:
        """获取当前价格所在的网格"""
        if price < self.lower_price:
            return -1
        if price >= self.upper_price:
            return self.grid_count
        
        return int((price - self.lower_price) / self.grid_spacing)
    
    def tick(self, current_price: float) -> list:
        """
        执行一次网格检查
        
        返回：执行的操作列表
        """
        current_grid = self.get_current_grid(current_price)
        operations = []
        
        # 检查每个网格
        for grid_price, state in self.grids.items():
            grid_idx = int((grid_price - self.lower_price) / self.grid_spacing)
            
            # 价格跌破网格 → 买入
            if current_price <= grid_price and not state["bought"]:
                # 计算可买入数量
                max_amount = self.quote_balance / grid_price
                amount = max_amount  # 全仓买入（简化）
                
                if amount > 0:
                    cost = amount * grid_price
                    self.quote_balance -= cost
                    self.base_balance += amount
                    state["bought"] = True
                    state["amount"] = amount
                    
                    operations.append({
                        "type": "BUY",
                        "grid": grid_idx,
                        "price": grid_price,
                        "amount": amount,
                        "cost": cost
                    })
                    self.trades.append(operations[-1])
        
        # 检查是否需要卖出（价格回到网格上方）
        for grid_price, state in self.grids.items():
            if current_price > grid_price and state["bought"]:
                # 卖出
                sell_amount = state["amount"]
                revenue = sell_amount * current_price
                profit = revenue - sell_amount * grid_price
                
                self.base_balance -= sell_amount
                self.quote_balance += revenue
                state["bought"] = False
                
                operations.append({
                    "type": "SELL",
                    "grid": int((grid_price - self.lower_price) / self.grid_spacing),
                    "price": current_price,
                    "amount": sell_amount,
                    "revenue": revenue,
                    "profit": profit
                })
                self.trades.append(operations[-1])
                self.profit_per_grid.append(profit)
        
        return operations
    
    def get_status(self, current_price: float) -> dict:
        """获取当前状态"""
        current_grid = self.get_current_grid(current_price)
        total_value = self.quote_balance + self.base_balance * current_price
        
        return {
            "current_price": current_price,
            "current_grid": current_grid,
            "quote_balance": self.quote_balance,
            "base_balance": self.base_balance,
            "total_value": total_value,
            "total_profit_pct": (total_value - self.initial_capital) / self.initial_capital * 100,
            "grids_active": sum(1 for g in self.grids.values() if g["bought"]),
            "total_trades": len(self.trades),
            "total_profit": sum(self.profit_per_grid)
        }
    
    def run_backtest(self, price_data: list) -> dict:
        """运行回测"""
        for price in price_data:
            self.tick(price)
        
        final_status = self.get_status(price_data[-1])
        
        return {
            "final_value": final_status["total_value"],
            "return_pct": final_status["total_profit_pct"],
            "total_trades": final_status["total_trades"],
            "grids_profitable": len([p for p in self.profit_per_grid if p > 0]),
            "grids_losing": len([p for p in self.profit_per_grid if p < 0]),
            "max_drawdown_pct": self.calculate_drawdown(price_data)
        }
    
    def calculate_drawdown(self, price_data: list) -> float:
        """计算最大回撤"""
        equity = []
        for price in price_data:
            equity.append(self.quote_balance + self.base_balance * price)
        
        peak = equity[0]
        max_dd = 0
        for e in equity:
            if e > peak:
                peak = e
            dd = (peak - e) / peak
            if dd > max_dd:
                max_dd = dd
        
        return max_dd * 100
```

```python
class MarketMakerStrategy:
    """
    简单做市策略
    
    策略逻辑：
    1. 在当前价格上下设置限价单
    2. 买入单在下方，卖出单在上方
    3. 价差（Spread）即利润
    4. 等待成交后，在对侧平仓
    """
    
    def __init__(self, spread_pct: float = 0.005, order_size: float = 0.1,
                 inventory_target: float = 0.5, initial_capital: float = 10000):
        """
        spread_pct: 买卖价差百分比（如 0.005 = 0.5%）
        order_size: 每笔订单数量
        inventory_target: 目标库存比例（0-1）
        """
        self.spread_pct = spread_pct
        self.order_size = order_size
        self.inventory_target = inventory_target
        self.initial_capital = initial_capital
        
        self.base_balance = 0
        self.quote_balance = initial_capital
        self.orders = []  # 活跃订单
        self.filled_orders = []
    
    def calculate_prices(self, mid_price: float) -> tuple:
        """计算买卖价格"""
        half_spread = mid_price * self.spread_pct / 2
        
        bid_price = mid_price - half_spread  # 买入价（我要买入）
        ask_price = mid_price + half_spread  # 卖出价（我要卖出）
        
        return bid_price, ask_price
    
    def calculate_inventory_ratio(self, current_price: float) -> float:
        """计算当前库存比例"""
        total_value = self.quote_balance + self.base_balance * current_price
        if total_value == 0:
            return 0
        
        base_value = self.base_balance * current_price
        return base_value / total_value
    
    def adjust_spread_for_inventory(self, base_spread: float, 
                                     inventory_ratio: float) -> float:
        """
        根据库存调整价差
        
        库存过多 → 扩大卖方价差，鼓励卖出
        库存过少 → 扩大买方价差，鼓励买入
        """
        target = self.inventory_target
        deviation = inventory_ratio - target
        
        # 库存偏离目标越多，价差调整越大
        adjustment = abs(deviation) * 0.02  # 最多调整 2%
        
        return base_spread + adjustment
    
    def place_orders(self, mid_price: float) -> list:
        """
        下单
        
        返回：新下的订单列表
        """
        inventory_ratio = self.calculate_inventory_ratio(mid_price)
        adaptive_spread = self.adjust_spread_for_inventory(
            self.spread_pct, inventory_ratio
        )
        
        bid_price, ask_price = self.calculate_prices(mid_price)
        
        # 根据库存决定下单方向
        new_orders = []
        
        # 库存低于目标 → 只挂买单
        if inventory_ratio < self.inventory_target:
            # 检查余额
            cost = self.order_size * bid_price
            if cost <= self.quote_balance:
                order = {
                    "side": "BUY",
                    "price": bid_price,
                    "amount": self.order_size,
                    "timestamp": len(self.filled_orders)  # 简化
                }
                self.orders.append(order)
                new_orders.append(order)
        
        # 库存高于目标 → 只挂卖单
        elif inventory_ratio > self.inventory_target:
            if self.base_balance >= self.order_size:
                order = {
                    "side": "SELL",
                    "price": ask_price,
                    "amount": self.order_size,
                    "timestamp": len(self.filled_orders)
                }
                self.orders.append(order)
                new_orders.append(order)
        
        # 库存接近目标 → 双边挂单
        else:
            # 买单
            cost = self.order_size * bid_price
            if cost <= self.quote_balance:
                order = {
                    "side": "BUY",
                    "price": bid_price,
                    "amount": self.order_size,
                    "timestamp": len(self.filled_orders)
                }
                self.orders.append(order)
                new_orders.append(order)
            
            # 卖单
            if self.base_balance >= self.order_size:
                order = {
                    "side": "SELL",
                    "price": ask_price,
                    "amount": self.order_size,
                    "timestamp": len(self.filled_orders)
                }
                self.orders.append(order)
                new_orders.append(order)
        
        return new_orders
    
    def check_fills(self, current_price: float) -> list:
        """
        检查订单是否成交
        
        返回：成交的订单列表
        """
        fills = []
        remaining = []
        
        for order in self.orders:
            if order["side"] == "BUY":
                # 买单成交：当前价格 <= 订单价格
                if current_price <= order["price"]:
                    # 执行买入
                    cost = order["amount"] * order["price"]
                    self.quote_balance -= cost
                    self.base_balance += order["amount"]
                    
                    fill = order.copy()
                    fill["fill_price"] = order["price"]
                    fills.append(fill)
                    self.filled_orders.append(fill)
                else:
                    remaining.append(order)
            
            elif order["side"] == "SELL":
                # 卖单成交：当前价格 >= 订单价格
                if current_price >= order["price"]:
                    revenue = order["amount"] * order["price"]
                    self.base_balance -= order["amount"]
                    self.quote_balance += revenue
                    
                    fill = order.copy()
                    fill["fill_price"] = order["price"]
                    fills.append(fill)
                    self.filled_orders.append(fill)
                else:
                    remaining.append(order)
        
        self.orders = remaining
        return fills
    
    def get_status(self, current_price: float) -> dict:
        """获取状态"""
        total_value = self.quote_balance + self.base_balance * current_price
        
        return {
            "total_value": total_value,
            "return_pct": (total_value - self.initial_capital) / self.initial_capital * 100,
            "quote_balance": self.quote_balance,
            "base_balance": self.base_balance,
            "inventory_ratio": self.calculate_inventory_ratio(current_price),
            "active_orders": len(self.orders),
            "total_fills": len(self.filled_orders),
            "spreads_earned": self.calculate_total_spreads()
        }
    
    def calculate_total_spreads(self) -> float:
        """计算累计获得的价差"""
        if len(self.filled_orders) < 2:
            return 0
        
        total_spread = 0
        buy_fills = [o for o in self.filled_orders if o["side"] == "BUY"]
        sell_fills = [o for o in self.filled_orders if o["side"] == "SELL"]
        
        # 简化计算：每对买卖单的平均价差
        pairs = min(len(buy_fills), len(sell_fills))
        for i in range(pairs):
            spread = sell_fills[i]["price"] - buy_fills[i]["price"]
            total_spread += spread
        
        return total_spread
    
    def run_backtest(self, price_data: list, rebalance_interval: int = 10) -> dict:
        """运行回测"""
        results = []
        
        for i, price in enumerate(price_data):
            # 检查成交
            fills = self.check_fills(price)
            
            # 定期重新挂单
            if i % rebalance_interval == 0:
                # 取消所有订单重新挂
                self.orders = []
                self.place_orders(price)
            
            results.append(self.get_status(price))
        
        final = results[-1]
        return {
            "final_value": final["total_value"],
            "return_pct": final["return_pct"],
            "total_fills": final["total_fills"],
            "total_spreads": final["spreads_earned"],
            "equity_curve": [r["total_value"] for r in results]
        }
```

```python
class CoveredCallStrategy:
    """
    备兑看涨期权策略（Covered Call）
    
    策略逻辑：
    1. 持有标的资产（ETH）
    2. 卖出虚值 Call 期权（收取权利金）
    3. 若价格上涨超过行权价 → 利润被限制
    4. 若价格下跌 → 权利金提供部分保护
    
    适合：预期横盘或小幅上涨
    """
    
    def __init__(self, initial_capital: float = 10000, target_premium_pct: float = 0.02):
        self.initial_capital = initial_capital
        self.target_premium_pct = target_premium_pct  # 目标权利金比例
        
        self.holdings = 0  # 持有的标的数量
        self.short_calls = []  # 卖出的 Call 合约
        self.capital = initial_capital
    
    def sell_covered_call(self, underlying_price: float, strike_price: float,
                          expiry_days: int, iv: float) -> dict:
        """
        卖出备兑看涨期权
        
        前提：已有标的持仓
        """
        from .option_pricing import BlackScholes
        
        T = expiry_days / 365
        r = 0.05
        
        # 计算期权价格
        call_price = BlackScholes.call_price(underlying_price, strike_price, T, r, iv)
        premium = call_price * underlying_price  # 权利金（按标的计价）
        
        # 下单
        order = {
            "type": "SELL_CALL",
            "underlying_price": underlying_price,
            "strike_price": strike_price,
            "premium": premium,
            "expiry_days": expiry_days,
            "iv": iv,
            "filled": True
        }
        
        self.short_calls.append(order)
        self.capital += premium  # 收取权利金
        
        return order
    
    def check_expiry(self, current_price: float, expiry_days: int = 0) -> dict:
        """
        检查期权到期
        
        返回：到期结算结果
        """
        if not self.short_calls:
            return {}
        
        result = {"exercised": False, "settlements": []}
        
        for call in self.short_calls:
            if expiry_days <= 0:  # 到期
                if current_price > call["strike_price"]:
                    # 被行权：卖出标的
                    result["exercised"] = True
                    result["settlements"].append({
                        "action": "DELIVER_UNDERLYING",
                        "strike": call["strike_price"],
                        "market_price": current_price,
                        "opportunity_cost": (current_price - call["strike_price"]) * self.holdings
                    })
                else:
                    # 到期作废：保留权利金
                    result["settlements"].append({
                        "action": "EXPIRE_WORTHLESS",
                        "premium_kept": call["premium"]
                    })
        
        self.short_calls = []  # 清除到期合约
        return result
    
    def calculate_max_profit(self, entry_price: float, strike_price: float,
                              premium: float) -> dict:
        """计算最大盈利和盈亏平衡点"""
        
        max_profit = (strike_price - entry_price + premium) * self.holdings
        breakeven = entry_price - premium
        
        return {
            "max_profit": max_profit,
            "max_profit_pct": max_profit / (entry_price * self.holdings) * 100,
            "breakeven": breakeven,
            "upside_capped_at": strike_price
        }
```

```python
class ArbitrageStrategy:
    """
    跨交易所统计套利
    
    策略逻辑：
    1. 监控两个交易所的价格差异
    2. 当价差 > 正常范围 → 买入低价的，卖空高价的
    3. 等待价差回归 → 平仓获利
    """
    
    def __init__(self, exchange_a: str, exchange_b: str,
                 z_score_threshold: float = 2.0,
                 lookback_period: int = 100,
                 initial_capital: float = 10000):
        self.exchange_a = exchange_a
        self.exchange_b = exchange_b
        self.z_score_threshold = z_score_threshold
        self.lookback_period = lookback_period
        self.initial_capital = initial_capital
        
        self.price_a_history = []
        self.price_b_history = []
        self.spread_history = []
        
        self.position = 0  # 1 = long A / short B, -1 = short A / long B
        self.entry_spread = 0
    
    def calculate_spread(self, price_a: float, price_b: float) -> float:
        """计算价差（百分比）"""
        return (price_b - price_a) / price_a
    
    def calculate_z_score(self) -> float:
        """计算当前价差的 Z-Score"""
        if len(self.spread_history) < self.lookback_period:
            return 0
        
        recent_spreads = self.spread_history[-self.lookback_period:]
        mean = np.mean(recent_spreads)
        std = np.std(recent_spreads)
        
        if std == 0:
            return 0
        
        current_spread = self.spread_history[-1]
        return (current_spread - mean) / std
    
    def signal(self, price_a: float, price_b: float) -> str:
        """
        生成交易信号
        """
        self.price_a_history.append(price_a)
        self.price_b_history.append(price_b)
        
        current_spread = self.calculate_spread(price_a, price_b)
        self.spread_history.append(current_spread)
        
        z_score = self.calculate_z_score()
        
        # 无持仓时
        if self.position == 0:
            # Z-Score 过高 → 买入 A，卖空 B（预期回归）
            if z_score > self.z_score_threshold:
                self.position = 1
                self.entry_spread = current_spread
                return "LONG_A_SHORT_B"
            
            # Z-Score 过低 → 卖空 A，买入 B（预期回归）
            if z_score < -self.z_score_threshold:
                self.position = -1
                self.entry_spread = current_spread
                return "SHORT_A_LONG_B"
        
        # 有持仓时
        else:
            # Z-Score 回归 → 平仓
            if abs(z_score) < 0.5:
                position = self.position
                self.position = 0
                pnl = (current_spread - self.entry_spread) * position
                return f"CLOSE_POSITION (PnL: {pnl:.4f})"
        
        return "HOLD"
    
    def run_backtest(self, prices_a: list, prices_b: list) -> dict:
        """运行回测"""
        if len(prices_a) != len(prices_b):
            raise ValueError("价格序列长度不一致")
        
        capital = self.initial_capital
        position = 0
        entry_spread = 0
        trades = []
        equity = []
        
        for i in range(len(prices_a)):
            signal = self.signal(prices_a[i], prices_b[i])
            
            if signal == "LONG_A_SHORT_B" and position == 0:
                position = 1
                entry_spread = self.spread_history[-1]
                trades.append({"type": "ENTER_LONG_A_SHORT_B", "spread": entry_spread})
            
            elif signal == "SHORT_A_LONG_B" and position == 0:
                position = -1
                entry_spread = self.spread_history[-1]
                trades.append({"type": "ENTER_SHORT_A_LONG_B", "spread": entry_spread})
            
            elif "CLOSE" in signal and position != 0:
                current_spread = self.spread_history[-1]
                pnl = (current_spread - entry_spread) * position
                capital += capital * pnl
                trades.append({"type": "CLOSE", "pnl": pnl})
                position = 0
            
            equity.append({
                "step": i,
                "spread": self.spread_history[-1],
                "capital": capital,
                "position": position
            })
        
        return {
            "final_capital": capital,
            "return_pct": (capital - self.initial_capital) / self.initial_capital * 100,
            "total_trades": len(trades),
            "winning_trades": len([t for t in trades if t.get("pnl", 0) > 0]),
            "win_rate": len([t for t in trades if t.get("pnl", 0) > 0]) / len(trades) * 100 if trades else 0,
            "max_drawdown_pct": self.calculate_max_dd(equity),
            "sharpe_ratio": self.calculate_sharpe([e["capital"] for e in equity])
        }
    
    @staticmethod
    def calculate_max_dd(equity: list) -> float:
        values = [e["capital"] for e in equity]
        peak = values[0]
        max_dd = 0
        for v in values:
            if v > peak:
                peak = v
            dd = (peak - v) / peak
            if dd > max_dd:
                max_dd = dd
        return max_dd * 100
    
    @staticmethod
    def calculate_sharpe(equity: list) -> float:
        if len(equity) < 2:
            return 0
        returns = np.diff(equity) / equity[:-1]
        return np.mean(returns) / np.std(returns) * np.sqrt(252) if np.std(returns) > 0 else 0
```

```python
class EventDrivenStrategy:
    """
    事件驱动策略
    
    策略逻辑：
    1. 监控链上事件（大户转账、合约交互）
    2. 监控市场事件（指数调整、期权到期）
    3. 根据事件类型和规模执行交易
    
    事件类型：
    - 大户转账（鲸鱼移动）
    - 合约解锁
    - 期权到期（Gamma Squeeze）
    - 指数再平衡
    """
    
    def __init__(self, initial_capital: float = 10000):
        self.initial_capital = initial_capital
        self.events = []
        self.position = 0
    
    def process_whale_transfer(self, from_address: str, to_address: str,
                                amount_usd: float, direction: str) -> dict:
        """
        处理大户转账事件
        
        direction: "inflow" = 流入 DEX, "outflow" = 流出到冷钱包
        """
        threshold = 1_000_000  # $1M 以上视为重要
        
        if amount_usd < threshold:
            return {"action": "IGNORE", "reason": "低于阈值"}
        
        event = {
            "type": "WHALE_TRANSFER",
            "from": from_address,
            "to": to_address,
            "amount_usd": amount_usd,
            "direction": direction
        }
        self.events.append(event)
        
        # 大量 USDT 流入 DEX → 可能准备抛售
        if "USDT" in from_address and direction == "inflow":
            return {"action": "BEARISH", "signal": " Whale 可能抛售"}
        
        # 大量 ETH 从交易所转出 → 看涨信号
        if "ETH" in from_address and direction == "outflow":
            return {"action": "BULLISH", "signal": " Whale 积累中"}
        
        return {"action": "NEUTRAL"}
    
    def process_token_unlock(self, token: str, unlock_amount_usd: float,
                              unlock_pct_of_supply: float) -> dict:
        """
        处理代币解锁事件
        """
        event = {
            "type": "TOKEN_UNLOCK",
            "token": token,
            "amount_usd": unlock_amount_usd,
            "pct_of_supply": unlock_pct_of_supply
        }
        self.events.append(event)
        
        # 解锁比例 > 5% → 短期抛压
        if unlock_pct_of_supply > 5:
            return {
                "action": "BEARISH",
                "signal": f"{token} 大规模解锁 {unlock_pct_of_supply:.1f}%，注意抛压",
                "strategy": "做空" if unlock_pct_of_supply > 10 else "减仓"
            }
        
        return {"action": "NEUTRAL"}
    
    def process_options_expiry(self, expiry_size_usd: float,
                                 put_call_ratio: float) -> dict:
        """
        处理期权到期事件
        
        put_call_ratio > 1 → 空头压力大，Gamma Squeeze 可能
        put_call_ratio < 1 → 多头压力大
        """
        event = {
            "type": "OPTIONS_EXPIRY",
            "size_usd": expiry_size_usd,
            "put_call_ratio": put_call_ratio
        }
        self.events.append(event)
        
        # 大额期权到期 + 高 Put/Call 比率 → Gamma Squeeze 风险
        if expiry_size_usd > 500_000_000 and put_call_ratio > 1.2:
            return {
                "action": "VOLATILE",
                "signal": "Gamma Squeeze 风险（大量 ITM Put 需对冲）",
                "strategy": "设置宽止损，防范剧烈波动"
            }
        
        # 大额期权到期 + 低 Put/Call 比率 → 上涨潜力
        if expiry_size_usd > 500_000_000 and put_call_ratio < 0.8:
            return {
                "action": "BULLISH",
                "signal": "看涨期权占主导，可能有买盘推动",
                "strategy": "逢低买入"
            }
        
        return {"action": "NEUTRAL"}
    
    def generate_event_signal(self) -> dict:
        """综合所有事件生成信号"""
        signals = {"bullish": 0, "bearish": 0, "neutral": 0}
        
        for event in self.events[-10:]:  # 最近10个事件
            if event["type"] == "WHALE_TRANSFER":
                pass  # 已处理
            elif event["type"] == "TOKEN_UNLOCK":
                if "BEARISH" in str(event):
                    signals["bearish"] += 1
                elif "BULLISH" in str(event):
                    signals["bullish"] += 1
        
        total = sum(signals.values())
        if total == 0:
            return {"overall": "NEUTRAL", "confidence": 0}
        
        dominant = max(signals, key=signals.get)
        confidence = signals[dominant] / total
        
        return {
            "overall": dominant.upper(),
            "confidence": confidence,
            "signals": signals,
            "recommendation": f"{dominant.upper()} 信号（置信度 {confidence:.0%}）"
        }
```

## 1. 均线交叉策略（MA Cross）

### 策略逻辑
短期均线上穿长期均线 → 买入；下穿 → 卖出。

### 参数边界
| 参数 | 合法范围 | 推荐值 | 说明 |
|------|---------|--------|------|
| MA短期 | 3 ~ 15 | 5, 8, 10 | 越小越灵敏，信号越多 |
| MA长期 | 10 ~ 60 | 20, 30, 60 | 越大越稳定，信号越少 |
| 最小差距 | 0 ~ 5% | 0（直接用交叉）| 防止震荡市假信号 |

```python
def signals_ma_cross(candles, fast=5, slow=20):
    from indicators import calc_sma
    prices = [c['close'] for c in candles]
    ma_fast = calc_sma(prices, fast)
    ma_slow = calc_sma(prices, slow)

    signals = []
    for i in range(1, len(candles)):
        if ma_fast[i] is None or ma_slow[i] is None:
            continue
        # 金叉买入
        if ma_fast[i] > ma_slow[i] and ma_fast[i-1] <= ma_slow[i-1]:
            signals.append({'type': 'BUY', 'index': i})
        # 死叉卖出
        elif ma_fast[i] < ma_slow[i] and ma_fast[i-1] >= ma_slow[i-1]:
            signals.append({'type': 'SELL', 'index': i})
    return signals
```

| 快线周期 | 8 ~ 20 | 12 | 默认 12 |
| 慢线周期 | 20 ~ 50 | 26 | 默认 26 |
| Signal | 5 ~ 15 | 9 | MACD 信号线周期 |

### 可执行代码
```python
def signals_macd(candles, fast=12, slow=26, signal=9):
    from indicators import calc_macd
    prices = [c['close'] for c in candles]
    macd_data = calc_macd(prices, fast, slow, signal)

    signals = []
    for i in range(1, len(candles)):
        dif = macd_data['macd'][i]
        dea = macd_data['signal'][i]
        dif_prev = macd_data['macd'][i-1]
        dea_prev = macd_data['signal'][i-1]
        if dif is None or dea is None:
            continue
        # 金叉：DIF从下方上穿DEA
        if dif_prev <= dea_prev and dif > dea:
            signals.append({'type': 'BUY', 'index': i})
        # 死叉：DIF从上方下穿DEA
        elif dif_prev >= dea_prev and dif < dea:
            signals.append({'type': 'SELL', 'index': i})
    return signals
```

### 风控建议
- DIF > 0 且金叉 → 胜率更高
- DIF < 0 且死叉 → 趋势向下跌幅更大

## 5. 资金费率套利（永续合约专项）

> 本策略为套利策略，不在 `backtest.py` 的技术分析回测范围内，
> 但可在 `paper_trade.py` 中手动模拟。

### 策略逻辑
做多永续合约 + 做空同品种现货 → 赚取资金费率
（只要资金费率 > 手续费，即为正期望）

### 完整 Python 实现
```python
def funding_arbitrage(strategy_name, candles, position_size=1000):
    """
    资金费率套利模拟
    position_size: 每边仓位 USD 价值
    """
    # 从 Binance 获取资金费率
    import urllib.request, json
    sym = 'BTCUSDT'
    url = f'https://fapi.binance.com/fapi/v1/premiumIndex?symbol={sym}'
    try:
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=8) as r:
            data = json.loads(r.read())
            funding_rate = float(data['lastFundingRate'])
            next_funding_time = int(data['nextFundingTime'])
    except Exception as e:
        print(f'无法获取资金费率: {e}')
        return None

    # 计算收益
    daily_settlements = 3
    maker_fee = 0.0002  # Maker手续费 0.02%

    # 每次结算净收益（双边各扣手续费）
    net_per_settlement = funding_rate - maker_fee * 2
    daily_return = net_per_settlement * daily_settlements
    annual_return = daily_return * 365

    result = {
        'symbol': sym,
        'funding_rate_pct': funding_rate * 100,
        'net_per_settlement_pct': net_per_settlement * 100,
        'daily_return_pct': daily_return * 100,
        'annual_return_pct': annual_return * 100,
        'position_size': position_size,
        'annual_profit': position_size * annual_return,
        'assessment': (
            'HIGH' if annual_return > 0.10 else
            'MEDIUM' if annual_return > 0.03 else
            'LOW'
        )
    }
    return result

# 示例
r = funding_arbitrage('funding', None)
if r:
    print(f"资金费率: {r['funding_rate_pct']:.4f}%")
    print(f"年化收益: {r['annual_return_pct']:.1f}%")
    print(f"评估: {r['assessment']}")
```

| 参数 | 说明 | 阈值 |

| 资金费率 | 每次结算费率（可在 Binance 官网查看）| 年化 > 10% 高机会 |
| 手续费 | Maker 约 0.02% | 需 < 资金费率 |

> 每个策略必须定义"什么情况下不能用"，否则等于没有风控

### 各策略失效条件

| 策略 | 适用行情 | 失效场景 | 退出条件 |
|------|---------|---------|----------|
| MA 交叉 | 趋势行情 | MA20走平（震荡市） | ADX < 20 连续5根K线 → 暂停 |
| MACD 突破 | 趋势行情 | 窄幅盘整 | 布林带宽度 < 3% → 暂停 |
| RSI 超买超卖 | 震荡行情 | 强趋势（RSI持续>70或<30） | RSI 连续8根K线>70 → 放弃该策略 |
| 布林带回归 | 区间震荡 | 趋势突破（价格沿轨道行走） | 收盘价连续3根K线在轨外 → 停用 |
| ADX+CCI 双确认 | 趋势确认 | ADX持续<20 | ADX < 20 连续10根K线 → 不入场 |
| KDJ 背离 | 趋势末期 | 强势单边（无背离） | 连续20根K线无背离信号 → 换策略 |
| CCI+OBV 量价 | 量价确认 | 低流动性（OBV平移） | 成交量 < 20日均量×0.5 → 不入场 |
| 资金费率套利 | 高费率期 | 费率回归正常 | Funding < 0.01% → 平仓退出 |

### 通用失效信号

无论使用哪个策略，出现以下任一条件时立即停用：

1. **连续5次止损** → 策略与当前行情不匹配，暂停1周
2. **月度 Sharpe < 0** → 策略在该月失效，下月再评估
3. **最大回撤超过策略历史回撤的1.5倍** → 触发回撤阶梯 YELLOW
4. **胜率连续30笔 < 35%** → 策略逻辑需要重新诊断

### 退出逻辑模板

```python
def should_exit_strategy(strategy_name, recent_trades, market_state):
    """判断策略是否应该退出"""
    # 1. 连续止损
    consecutive_losses = 0
    for t in reversed(recent_trades):
        if t['pnl'] < 0:
            consecutive_losses += 1
        else:
            break
    if consecutive_losses >= 5:
        return True, f'连续{consecutive_losses}次止损，暂停策略'
    
    # 2. 近30笔胜率
    recent_30 = recent_trades[-30:]
    if len(recent_30) >= 20:
        win_rate = sum(1 for t in recent_30 if t['pnl'] > 0) / len(recent_30)
        if win_rate < 0.35:
            return True, f'近{len(recent_30)}笔胜率{win_rate:.0%}，策略失效'
    
    # 3. 市场状态不匹配
    if strategy_name in ('ma_cross', 'macd') and market_state == 'sideways':
        return True, '趋势策略不适用震荡市'
    if strategy_name in ('rsi', 'bollinger') and market_state == 'strong_trend':
        return True, '震荡策略不适用强趋势'
    
    return False, '策略正常运行'
```

## 1. 双均线策略（MA Crossover）

### 策略逻辑
- **买入信号**：短周期 MA 上穿长周期 MA（金叉）
- **卖出信号**：短周期 MA 下穿长周期 MA（死叉）
- **适用市场**：趋势行情（牛市/熊市主浪）
- **震荡市表现**：差（频繁假信号）

### 参数说明
| 参数 | 默认值 | 说明 |
|------|--------|------|
| `fast` | 20 | 短周期（小时线建议 20，日线建议 5-10） |
| `slow` | 60 | 长周期（小时线建议 60，日线建议 20-30） |
| `ma_type` | `sma` | `sma` 或 `ema` |

### 回测代码（可直接运行）
```python
# backtest_ma_cross.py - 双均线策略回测
import sys
sys.path.insert(0, 'scripts')

from indicators import calc_sma, calc_ema
from backtest import BacktestEngine

def ma_cross_strategy(candles, fast=20, slow=60, ma_type='sma'):
    """
    双均线策略
    返回: [None, 'BUY', 'SELL', None, ...]
    """
    closes = [c['close'] for c in candles]
    
    if ma_type == 'sma':
        fast_line = calc_sma(closes, fast)
        slow_line = calc_sma(closes, slow)
    else:
        fast_line = calc_ema(closes, fast)
        slow_line = calc_ema(closes, slow)
    
    signals = [None] * len(closes)
    position = None  # None, 'LONG'
    
    for i in range(1, len(closes)):
        if fast_line[i] is None or slow_line[i] is None:
            continue
        
        # 金叉：短均线上穿长均线
        if fast_line[i-1] <= slow_line[i-1] and fast_line[i] > slow_line[i]:
            if position is None:
                signals[i] = 'BUY'
                position = 'LONG'
        
        # 死叉：短均线下穿长均线
        elif fast_line[i-1] >= slow_line[i-1] and fast_line[i] < slow_line[i]:
            if position == 'LONG':
                signals[i] = 'SELL'
                position = None
    
    return signals

# 运行回测

    )
    
    # 加载数据（自动从 API 或本地 CSV）
```

## 3. 布林带突破策略

### 策略逻辑
- **买入信号**：价格触及下轨后回到中轨上方
- **卖出信号**：价格触及上轨后回到中轨下方
- **适用市场**：震荡市
- **趋势市表现**：中等

### 回测代码
```python
# backtest_bollinger.py
import sys
sys.path.insert(0, 'scripts')

from indicators import calc_bollinger_bands
from backtest import BacktestEngine

def bollinger_strategy(candles, period=20, std_dev=2.0):
    closes = [c['close'] for c in candles]
    bb = calc_bollinger_bands(candles, period=period, std_dev=std_dev)
    
    signals = [None] * len(closes)
    position = None
    touched_lower = False
    
    for i in range(len(closes)):
        if bb['lower'][i] is None:
            continue
        
        # 触及下轨
        if closes[i] <= bb['lower'][i]:
            touched_lower = True
        
        # 回到中轨上方 → 买入
        if touched_lower and closes[i] > bb['middle'][i] and position is None:
            signals[i] = 'BUY'
            position = 'LONG'
            touched_lower = False
        
        # 触及上轨 → 卖出
        if closes[i] >= bb['upper'][i] and position == 'LONG':
            signals[i] = 'SELL'
            position = None
    
    return signals
```

## 4. 资金费率套利策略（Web3 独有）

### 策略逻辑
- **做多现货 + 做空永续合约**：赚取资金费率
- **当 funding rate > 0**：永续合约空头每天收取资金费（通常 8 小时一次）
- **适用市场**：牛市（funding rate 长期为正）
- **2024 年 BTC 平均 funding rate**：+0.01%/8h = +11% 年化

### 回测代码
```python
# backtest_funding_arb.py - 资金费率套利
import sys
sys.path.insert(0, 'scripts')

from indicators import calc_funding_signal
from data_fetch import fetch_funding_rates

def funding_arb_strategy(symbol, lookback=24):
    """
    资金费率套利：
    - funding_rate > 0.01% → 做空永续，做多现货（对冲 delta）
    - funding_rate < -0.01% → 做多永续，做空现货
    """
    # 获取历史 funding rates
    rates = fetch_funding_rates(symbol, limit=lookback)
    
    signals = calc_funding_signal(rates, threshold_high=0.01, threshold_low=-0.01)
    
    # signals: ['LONG_ARB', 'SHORT_ARB', None, ...]
    return signals
```

## 5. MACD 金叉死叉策略

### 策略逻辑
- **买入**：MACD 线上穿 Signal 线（金叉）
- **卖出**：MACD 线下穿 Signal 线（死叉）
- **适用**：趋势行情

### 回测代码
```python
# backtest_macd.py
import sys
sys.path.insert(0, 'scripts')

from indicators import calc_macd
from backtest import BacktestEngine

def macd_strategy(candles, fast=12, slow=26, signal=9):
    closes = [c['close'] for c in candles]
    macd = calc_macd(closes, fast, slow, signal)
    
    signals = [None] * len(closes)
    position = None
    
    for i in range(1, len(closes)):
        if macd['macd'][i] is None or macd['signal'][i] is None:
            continue
        
        # 金叉
        if macd['macd'][i-1] <= macd['signal'][i-1] and macd['macd'][i] > macd['signal'][i]:
            if position is None:
                signals[i] = 'BUY'
                position = 'LONG'
        
        # 死叉
        elif macd['macd'][i-1] >= macd['signal'][i-1] and macd['macd'][i] < macd['signal'][i]:
            if position == 'LONG':
                signals[i] = 'SELL'
                position = None
    
    return signals
```

## 6. 多因子组合策略

### 策略逻辑
组合多个因子，加权打分，只有总分超过阈值才开仓：

```
综合得分 = 0.3*RSI信号 + 0.3*MACD信号 + 0.2*布林带信号 + 0.2*OBV信号
买入：综合得分 > 0.6
卖出：综合得分 < 0.4
```

### 回测代码
```python

                       calc_obv, calc_ema)
from backtest import BacktestEngine

def multi_factor_strategy(candles):
    closes = [c['close'] for c in candles]
    
    # 计算各因子信号（归一化到 [-1, 1]）
    rsi = calc_rsi(closes, 14)
    macd = calc_macd(closes, 12, 26, 9)
    bb = calc_bollinger_bands(candles, 20, 2.0)
    obv = calc_obv(candles)
    
    # 因子打分（示例）
    scores = [0.0] * len(closes)
    
    for i in range(len(closes)):
        if rsi[i] is None:
            continue
        
        # RSI 因子
        rsi_score = (70 - rsi[i]) / 40  # RSI=30 → 1.0, RSI=70 → 0.0
        rsi_score = max(0, min(1, rsi_score))
        
        # MACD 因子
        macd_score = 0.5
        if macd['macd'][i] is not None and macd['signal'][i] is not None:
            macd_score = 1.0 if macd['macd'][i] > macd['signal'][i] else 0.0
        
        # 综合得分
        scores[i] = 0.3 * rsi_score + 0.3 * macd_score  # + ...
    
    # 生成信号
    signals = [None] * len(closes)
    position = None
    for i in range(len(scores)):
        if scores[i] > 0.6 and position is None:
            signals[i] = 'BUY'
            position = 'LONG'
        elif scores[i] < 0.4 and position == 'LONG':
            signals[i] = 'SELL'
            position = None
    
    return signals
```

## 7. Funding Rate 永续合约套利（详细版）

### Web3 独有策略详解

资金费率（Funding Rate）是永续合约市场的独特机制：
- 当市场看多 → funding rate > 0 → 多头支付给空头
- 当市场看空 → funding rate < 0 → 空头支付给多头
- 结算周期：每 8 小时一次（Binance/Bybit）

### 套利方式
```
Delta 中性套利：
  做多现货 BTC（1x）
  做空永续 BTC（1x）
  → 赚取 funding rate，不承担 BTC 价格风险
```

### 回测框架
```python
# funding_arb_backtest.py
import sys
sys.path.insert(0, 'scripts')

from data_fetch import fetch_funding_rates, fetch_klines
from backtest import calculate_pnl

def backtest_funding_arb(symbol, start, end, capital=10000):
    """
    回测资金费率套利策略
    
    收益来源：
      1. Funding rate 收入（多头支付给空头）
      2. 基差收益（现货和合约价格差）
    
    成本：
      1. 开仓手续费（现货 0.1% + 合约 0.02%）
      2. 资金占用成本（保证金）
    """
    # 获取 funding rates 历史数据
    rates = fetch_funding_rates(symbol, start=start, end=end)
    
    # 获取现货和合约价格
    spot_prices = fetch_klines(symbol.replace('USDT', 'USDT'), start, end)
    perp_prices = fetch_klines(symbol + 'PERP', start, end)
    
    total_funding_income = 0
    total_fee_cost = 0
    
    for i, rate in enumerate(rates):
        # 每 8 小时结算一次
        funding_income = capital * rate  # rate 通常为 0.0001（0.01%）
        total_funding_income += funding_income
    
    # 估算手续费（开仓 + 平仓）
    total_fee_cost = capital * 0.001 * 2  # 0.1% * 2（开+平）
    
    net_profit = total_funding_income - total_fee_cost
    annual_return = net_profit / capital * (365 / len(rates))
    
    return {
        'total_funding_income': total_funding_income,
        'total_fee_cost': total_fee_cost,
        'net_profit': net_profit,
        'annual_return': annual_return,
    }

# 运行示例
if __name__ == '__main__':
    result = backtest_funding_arb(
        symbol='BTCUSDT',
        start='2024-01-01',
        end='2024-12-31',
        capital=10000
    )
    print(f"总收益: ${result['net_profit']:.2f}")
    print(f"年化收益: {result['annual_return']*100:.2f}%")
```

## 8. OBV 量价背离策略

### 策略逻辑
- **买入信号**：价格创新低，但 OBV 不创新低（底背离）
- **卖出信号**：价格创新高，但 OBV 不创新高（顶背离）
- **适用**：趋势反转识别

### 回测代码
```python
# backtest_obv_divergence.py
import sys
sys.path.insert(0, 'scripts')

from indicators import calc_obv
from backtest import BacktestEngine

def obv_divergence_strategy(candles, lookback=20):
    closes = [c['close'] for c in candles]
    obv = calc_obv(candles)
    
    signals = [None] * len(closes)
    position = None
    
    for i in range(lookback, len(closes)):
        # 价格新低
        price_low = min(closes[i-lookback:i+1])
        price_new_low = closes[i] == price_low
        
        # OBV 不创新低（底背离）
        obv_low = min(obv[i-lookback:i+1])
        obv_not_new_low = obv[i] > obv_low
        
        if price_new_low and obv_not_new_low and position is None:
            signals[i] = 'BUY'
            position = 'LONG'
        
        # 顶背离
        price_high = max(closes[i-lookback:i+1])
        price_new_high = closes[i] == price_high
        obv_high = max(obv[i-lookback:i+1])
        obv_not_new_high = obv[i] < obv_high
        
        if price_new_high and obv_not_new_high and position == 'LONG':
            signals[i] = 'SELL'
            position = None
    
    return signals
```

```python
def apply_slippage(price, side, volatility):
    """根据波动率动态计算滑点"""
    base_slippage = 0.0005  # 0.05%
    vol_multiplier = volatility * 10  # 高波动时滑点更大
    slippage = base_slippage * (1 + vol_multiplier)
    
    if side == 'BUY':
        return price * (1 + slippage)
    else:
        return price * (1 - slippage)
```
