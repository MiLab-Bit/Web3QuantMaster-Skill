def estimate_airdrop_potential(protocol: str) -> dict:
    """
    评估协议的潜在空投规模
    
    评估因素：
    1. 是否已有代币
    2. VC 融资规模和估值
    3. 是否已有空投公告
    4. 生态规模（TVL、用户数）
    5. 竞争对手是否已发币
    """
    data = get_protocol_data(protocol)
    
    # 评估框架
    score = 0
    
    # 1. 无代币（0 = 已有代币）
    if not has_token(protocol):
        score += 30  # 未发币 = 空投潜力大
    
    # 2. VC 融资
    vc_raised = data.get("vc_raised", 0)
    if vc_raised > 100_000_000:
        score += 20  # 高估值 VC = 大规模空投可能性高
    elif vc_raised > 10_000_000:
        score += 15
    
    # 3. 生态规模
    tvl = data.get("tvl", 0)
    if tvl > 1_000_000_000:
        score += 25
    elif tvl > 100_000_000:
        score += 15
    elif tvl > 10_000_000:
        score += 10
    
    # 4. 用户数
    users = data.get("unique_users", 0)
    if users > 100_000:
        score += 15
    elif users > 10_000:
        score += 10
    elif users > 1000:
        score += 5
    
    # 5. 竞争对手已发币
    if has_competitor_token(protocol):
        score += 10  # 竞争对手已发 = 该协议更可能跟进
    
    return {
        "protocol": protocol,
        "score": score,
        "rating": "EXCELLENT" if score >= 80 else "GOOD" if score >= 60 else "MEDIUM" if score >= 40 else "LOW",
        "estimated_airdrop_size": "$500-5000" if score >= 60 else "$100-500" if score >= 40 else "不确定",
        "priority": "HIGH" if score >= 60 else "MEDIUM" if score >= 40 else "LOW"
    }
