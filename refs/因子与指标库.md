## 一、协议收入因子

### 核心指标

协议收入是 DeFi 协议健康度的核心指标：

```python
def calc_protocol_revenue_factors(protocol_address: str, web3) -> dict:
    """
    计算 DeFi 协议的收入因子
    
    主要指标：
    - Total Revenue: 协议总收入
    - Protocol Revenue: 协议留存收入（vs. 支付给 LP/用户的收入）
    - Fee: 用户支付的交易费用
    - Volume: 协议处理的交易量
    """
    # 从合约或 subgraph 获取数据
    # 简化实现
    
    total_revenue_24h = fetch_event_sum(protocol_address, "Revenue", hours=24)
    protocol_revenue_24h = fetch_event_sum(protocol_address, "ProtocolRevenue", hours=24)
    volume_24h = fetch_volume(protocol_address, hours=24)
    
    # Fee = Total Revenue / Volume
    fee_rate = total_revenue_24h / volume_24h if volume_24h > 0 else 0
    
    return {
        "total_revenue_24h": total_revenue_24h,
        "protocol_revenue_24h": protocol_revenue_24h,
        "volume_24h": volume_24h,
        "fee_rate": fee_rate,  # 协议费率
        "revenue_per_token": protocol_revenue_24h / fetch_token_supply(protocol_address),
        "revenue_growth_7d": calc_growth_rate(total_revenue_24h, days=7)
    }

def fetch_event_sum(contract_address, event_name, hours=24):
    """从合约事件日志中计算总收入"""
    # 实际实现：从合约的 Profit 事件中聚合
    # 这里返回模拟数据
    import random
    import time
    
    base_revenue = 100000  # $100k 基础收入

def fetch_volume(protocol, hours=24):
    """获取协议交易量"""
    # 从 DeFiLlama 或 The Graph 获取
    return 50000000 * hours / 24  # 模拟：$50M/天
```

### 收入因子公式

| 因子名 | 公式 | 含义 |
|--------|------|------|
| **Protocol Revenue** | Σ(每笔交易费用 × 协议抽成比例) | 协议直接收入 |
| **Fee Per Share** | Protocol Revenue / Total Shares | 每份份额的收入 |
| **Revenue Growth Rate** | (Rev_t - Rev_{t-7}) / Rev_{t-7} | 收入增长趋势 |
| **Revenue / FDV** | Annual Revenue / Fully Diluted Valuation | 市盈率等价指标 |

```python
def revenue_factors_score(factors: dict) -> dict:
    """
    综合评分：协议收入因子
    分数范围：0-100
    """
    scores = {}
    
    # 收入增长率评分
    growth = factors.get("revenue_growth_7d", 0)
    if growth > 0.5:  # > 50% 周增长
        scores["growth"] = 90
    elif growth > 0.2:
        scores["growth"] = 70
    elif growth > 0:
        scores["growth"] = 50
    else:
        scores["growth"] = 30
    
    # 收入稳定性评分（基于标准差）
    # 实际实现需要历史数据
    scores["stability"] = 70  # 简化
    
    # 协议留存率（Protocol Revenue / Total Revenue）
    retention = factors.get("protocol_revenue_24h", 0) / max(factors.get("total_revenue_24h", 1), 1)
    scores["retention"] = retention * 100
    
    # 综合得分
    weights = {"growth": 0.4, "stability": 0.3, "retention": 0.3}
    total_score = sum(scores[k] * weights[k] for k in weights)
    
    return {
        "scores": scores,
        "total_score": total_score,
        "rating": "A" if total_score > 80 else "B" if total_score > 60 else "C"
    }
```

```python
def calc_dex_liquidity_factors(pool_address: str, web3) -> dict:
    """
    计算 DEX 池子的流动性因子
    
    核心指标：
    - TVL (Total Value Locked): 池子总锁仓量
    - Volume / TVL Ratio: 周转率
    - Utilization Rate: 利用率
    """
    # 获取池子数据
    pool_data = fetch_pool_data(pool_address)
    
    tvl = pool_data["reserve0_usd"] + pool_data["reserve1_usd"]
    volume_24h = pool_data["volume_24h"]
    
    # 周转率 = Volume / TVL
    turnover = volume_24h / tvl if tvl > 0 else 0
    
    # 利用率 = 已借出 / 可借（对于借贷协议）
    # 对于 DEX：交易深度 / TVL
    utilization = volume_24h / (tvl * 10) if tvl > 0 else 0  # 假设 TVL 的 10% 参与交易
    
    # 流动性集中度（V3 特有）
    if pool_data.get("is_v3"):
        concentration = calc_liquidity_concentration(pool_address)
    else:
        concentration = 1.0
    
    return {
        "tvl": tvl,
        "volume_24h": volume_24h,
        "turnover_24h": turnover,
        "turnover_7d": turnover * 7,
        "concentration": concentration,  # V3 集中度
        "liquidity_score": calc_liquidity_score(tvl, turnover),
        "stability_7d": calc_tvl_stability(pool_address, days=7)
    }

def calc_liquidity_score(tvl, turnover):
    """
    流动性综合评分
    基于 TVL 和周转率
    """
    # TVL 评分（对数刻度）
    if tvl > 100_000_000:  # > $100M
        tvl_score = 90
    elif tvl > 10_000_000:  # > $10M
        tvl_score = 70
    elif tvl > 1_000_000:   # > $1M
        tvl_score = 50
    else:
        tvl_score = 30
    
    # 周转率评分
    if turnover > 1.0:  # 日周转 > 100%
        turn_score = 90
    elif turnover > 0.5:
        turn_score = 70
    elif turnover > 0.1:
        turn_score = 50
    else:
        turn_score = 30
    
    # 综合评分
    return (tvl_score * 0.6 + turn_score * 0.4)

def calc_liquidity_concentration(pool_address):
    """
    计算 Uniswap V3 的流动性集中度
    集中度 = 80% 流动性所在的价格区间宽度
    """
    # 获取所有 tick 的流动性
    ticks = fetch_v3_ticks(pool_address)
    
    # 计算累计流动性
    liquidity_depth = {}
    for tick in ticks:
        liquidity_depth[tick["price"]] = tick["liquidity"]
    
    total_liquidity = sum(liquidity_depth.values())
    sorted_prices = sorted(liquidity_depth.keys())
    
    # 找出 80% 流动性所在的价格区间
    cumsum = 0
    low_price = sorted_prices[0]
    high_price = sorted_prices[-1]
    
    for price in sorted_prices:
        cumsum += liquidity_depth[price]
        if cumsum < total_liquidity * 0.1:
            low_price = price
        if cumsum > total_liquidity * 0.9:
            high_price = price
            break
    
    mid_price = (sorted_prices[0] + sorted_prices[-1]) / 2
    concentration_range = (high_price - low_price) / mid_price
    
    return concentration_range  # 越小越集中
```

```python
def calc_lending_liquidity_factors(protocol_address: str) -> dict:
    """
    计算借贷协议的流动性因子
    
    核心指标：
    - Total Supplied: 总存款
    - Total Borrowed: 总借款
    - Utilization Rate: 利用率 = Borrowed / Supplied
    - Liquidation Risk: 清算风险
    """
    # 获取借贷数据
    supplied = fetch_lending_metric(protocol_address, "totalSupplied")
    borrowed = fetch_lending_metric(protocol_address, "totalBorrowed")
    reserves = fetch_lending_metric(protocol_address, "reserves")
    
    # 利用率
    utilization = borrowed / supplied if supplied > 0 else 0
    
    # 可用流动性
    available_liquidity = supplied - borrowed
    
    # 清算阈值
    liquidation_threshold = 0.8  # 抵押率 80%，清算线 80%
    
    # 预估清算量（假设全部抵押品）
    at_risk = borrowed * (utilization > liquidation_threshold)
    
    return {
        "total_supplied": supplied,
        "total_borrowed": borrowed,
        "utilization": utilization,
        "utilization_pct": utilization * 100,
        "available_liquidity": available_liquidity,
        "liquidation_risk": "HIGH" if utilization > 0.85 else "MEDIUM" if utilization > 0.7 else "LOW",
        "reserves_ratio": reserves / borrowed if borrowed > 0 else 0,
        "at_risk_amount": at_risk
    }
```

```python
def calc_user_behavior_factors(protocol_address: str) -> dict:
    """
    计算用户行为因子
    
    核心指标：
    - MAU (Monthly Active Users): 月活跃用户
    - Active Wallets Growth: 活跃钱包增长
    - Retention Rate: 留存率
    - Whale Ratio: 鲸鱼占比
    """
    # 获取用户数据
    transactions = fetch_transactions(protocol_address, days=30)
    
    # 唯一钱包数
    unique_wallets = set(tx["from"] for tx in transactions) | set(tx["to"] for tx in transactions)
    mau = len(unique_wallets)
    
    # 上月数据（对比）
    prev_mau = 1000  # 简化
    mau_growth = (mau - prev_mau) / prev_mau if prev_mau > 0 else 0
    
    # 鲸鱼占比（持仓 > $100k 的钱包）
    whale_threshold_usd = 100000
    whale_count = 0
    whale_volume = 0
    total_volume = sum(tx["value_usd"] for tx in transactions)
    
    for wallet in unique_wallets:
        balance = get_wallet_balance(wallet, protocol_address)
        if balance > whale_threshold_usd:
            whale_count += 1
            whale_volume += balance
    
    whale_ratio = whale_count / mau if mau > 0 else 0
    whale_volume_ratio = whale_volume / total_volume if total_volume > 0 else 0
    
    # 交易频率
    avg_tx_per_wallet = len(transactions) / mau if mau > 0 else 0
    
    return {
        "mau": mau,
        "mau_growth": mau_growth,
        "mau_growth_pct": mau_growth * 100,
        "whale_count": whale_count,
        "whale_ratio": whale_ratio,
        "whale_volume_ratio": whale_volume_ratio,
        "avg_tx_per_wallet": avg_tx_per_wallet,
        "engagement_score": calc_engagement_score(mau, whale_ratio, avg_tx_per_wallet)
    }

def calc_engagement_score(mau, whale_ratio, avg_tx):
    """用户参与度评分"""
    # MAU 评分
    mau_score = min(mau / 10000 * 100, 100)  # 10k 用户 = 100分
    
    # 鲸鱼占比评分（低鲸鱼占比 = 更去中心化 = 更高分）
    whale_score = max(100 - whale_ratio * 200, 0)  # 0% 鲸鱼 = 100分，50%+ = 0分
    
    # 交易频率评分
    tx_score = min(avg_tx * 20, 100)  # 5笔/月 = 100分
    
    weights = {"mau": 0.4, "whale": 0.3, "tx": 0.3}
    return mau_score * weights["mau"] + whale_score * weights["whale"] + tx_score * weights["tx"]
```

```python
def calc_contract_interaction_factors(protocol_address: str, user_address: str) -> dict:
    """
    计算用户与协议的交互因子
    用于量化用户对协议的依赖度
    """
    # 获取用户在协议上的历史交互
    interactions = fetch_user_interactions(user_address, protocol_address, days=90)
    
    if not interactions:
        return {"interaction_score": 0, "recommendation": "新用户，无历史数据"}
    
    # 交互频率
    freq = len(interactions) / 90  # 每天交互次数
    
    # 交互深度（平均交易额）
    avg_value = sum(tx["value_usd"] for tx in interactions) / len(interactions)
    
    # 交互多样性（调用的函数种类）
    unique_functions = set(tx["function"] for tx in interactions)
    diversity = len(unique_functions) / 10  # 假设 10 种核心函数
    
    # 时间衰减权重（近期交互更重要）
    decay_factor = calc_time_decay(interactions)
    
    # 综合评分
    score = (
        min(freq * 20, 30) +       # 频率（最多30分）
        min(avg_value / 10000 * 30, 30) +  # 金额（最多30分）
        min(diversity * 20, 20) +   # 多样性（最多20分）
        decay_factor * 20           # 时间衰减（最多20分）
    )
    
    return {
        "interaction_count": len(interactions),
        "avg_value_usd": avg_value,
        "function_diversity": len(unique_functions),
        "interaction_score": score,
        "recommendation": "HOLD" if score > 60 else "REDUCE" if score > 30 else "CLOSE"
    }
```

```python
def calc_protocol_security_factors(protocol_address: str) -> dict:
```

```python
def calc_governance_factors(protocol_address: str) -> dict:
    """
    计算治理因子
    
    核心指标：
    - 治理代币持有分布
    - 提案参与率
    - 提案通过率
    - 国库余额
    """
    # 获取治理数据
    token_holders = fetch_token_holders(protocol_address)
    proposals = fetch_proposals(protocol_address, days=180)
    
    # 1. 代币分布（基尼系数）
    holdings = [h["balance"] for h in token_holders]
    gini = calc_gini_coefficient(holdings)
    
    # 2. Top 10 持有占比
    top10_share = sum(holdings[:10]) / sum(holdings) if holdings else 0
    
    # 3. 提案参与率
    if proposals:
        avg_participation = sum(p["votes"] for p in proposals) / len(proposals) / sum(holdings)
    else:
        avg_participation = 0
    
    # 4. 提案通过率
    passed = sum(1 for p in proposals if p["status"] == "Passed")
    pass_rate = passed / len(proposals) if proposals else 0
    
    # 5. 国库余额
    treasury = get_treasury_balance(protocol_address)
    
    # 综合治理评分
    governance_score = (
        (100 - top10_share * 100) * 0.3 +  # Top10 持有越低越好
        min(avg_participation * 1000, 50) * 0.2 +  # 参与率
        pass_rate * 100 * 0.2 +  # 通过率
        (100 if treasury > 10_000_000 else 50 if treasury > 1_000_000 else 0) * 0.3  # 国库
    )
    
    return {
        "top10_share": top10_share,
        "gini_coefficient": gini,
        "avg_participation_rate": avg_participation,
        "proposal_pass_rate": pass_rate,
        "treasury_balance_usd": treasury,
        "governance_score": governance_score,
        "governance_rating": "A" if governance_score > 70 else "B" if governance_score > 50 else "C"
    }

def calc_gini_coefficient(values):
    """计算基尼系数"""
    if not values:
        return 0
    
    sorted_values = sorted(values)
    n = len(values)
    cumsum = sum((i + 1) * v for i, v in enumerate(sorted_values))
    total = sum(sorted_values)
    
    if total == 0:
        return 0
    
    return (2 * cumsum) / (n * total) - (n + 1) / n
```

## 六、跨协议相关性

```python
def calc_cross_protocol_factors(portfolio_protocols: list) -> dict:
    """
    计算跨协议相关性
    用于评估组合的协议集中度风险
    """
    # 获取各协议的收益率
    returns = {}
    for protocol in portfolio_protocols:
        returns[protocol] = fetch_protocol_returns(protocol, days=30)
    
    # 计算相关性矩阵
    protocol_names = list(returns.keys())
    corr_matrix = {}
    
    for i, p1 in enumerate(protocol_names):
        corr_matrix[p1] = {}
        for j, p2 in enumerate(protocol_names):
            if i == j:
                corr_matrix[p1][p2] = 1.0
            elif j < i:
                corr_matrix[p1][p2] = corr_matrix[p2][p1]
            else:
                corr_matrix[p1][p2] = calc_correlation(returns[p1], returns[p2])
    
    # 找出高相关性协议对（分散风险）
    high_corr_pairs = []
    for p1 in protocol_names:
        for p2 in protocol_names:
            if p1 < p2 and corr_matrix[p1][p2] > 0.7:
                high_corr_pairs.append((p1, p2, corr_matrix[p1][p2]))
    
    # 平均相关性（衡量组合集中度）
    n = len(protocol_names)
    total_corr = sum(corr_matrix[p1][p2] for i, p1 in enumerate(protocol_names) for j, p2 in enumerate(protocol_names) if i < j)
    avg_corr = total_corr / (n * (n - 1) / 2) if n > 1 else 0
    
    return {
        "correlation_matrix": corr_matrix,
        "high_correlation_pairs": high_corr_pairs,
        "average_correlation": avg_corr,
        "concentration_risk": "HIGH" if avg_corr > 0.6 else "MEDIUM" if avg_corr > 0.4 else "LOW",
        "diversification_score": (1 - avg_corr) * 100
    }

def calc_correlation(series1, series2):
    """计算皮尔逊相关系数"""
    import numpy as np
    
    if len(series1) != len(series2) or len(series1) < 2:
        return 0
    
    return np.corrcoef(series1, series2)[0, 1]
```

```python
def screen_protocols(protocols: list, min_score=60) -> list:
    """
    基于多因子筛选优质 DeFi 协议
    """
    results = []
    
    for protocol in protocols:
        revenue = calc_protocol_revenue_factors(protocol)
        liquidity = calc_dex_liquidity_factors(protocol)
        security = calc_protocol_security_factors(protocol)
        
        # 综合评分

            revenue_factors_score(revenue)["total_score"] * 0.3 +
            liquidity["liquidity_score"] * 0.3 +
            security["security_score"] * 0.4
        )
        
        if total_score >= min_score:
            results.append({
                "protocol": protocol,
                "total_score": total_score,
                "revenue_score": revenue_factors_score(revenue)["total_score"],
                "liquidity_score": liquidity["liquidity_score"],
                "security_score": security["security_score"]
            })
    
    return sorted(results, key=lambda x: x["total_score"], reverse=True)
```

## 2. 交易所因子

### 因子 1：Funding Rate（资金费率）
```
定义：永续合约资金费率（8h 一次）
计算：直接读取交易所 API
标准化：z-score((funding_rate - mean) / std)
```

**策略信号**：
- `Funding Rate > 0.02%` → 市场看多过度 → 做空信号
- `Funding Rate < -0.01%` → 市场看空过度 → 做多信号

**IC 分析（2024 年 BTC）**：
| 期限 | IC | IC_IR | 说明 |
|------|----|-------|------|
| 1d | 0.08 | 0.35 | 弱信号 |
| 3d | 0.12 | 0.42 | 中等信号 |
| 7d | 0.15 | 0.48 | 强信号 |

> **结论**：Funding Rate 是**反向因子**（高费率 → 未来收益低），适合均值回归策略。

**代码模板**：
```python
# fetch_funding_factor.py
import numpy as np

def calculate_funding_factor(symbol, lookback=30):
    """
    计算 Funding Rate 因子（z-score）
    """
    rates = fetch_funding_rates(symbol, limit=lookback*3)  # 每天 3 次
    rates_daily = [np.mean(rates[i:i+3]) for i in range(0, len(rates), 3)]
    
    current = rates_daily[-1]
    mean = np.mean(rates_daily)
    std = np.std(rates_daily)
    
    z_score = (current - mean) / std if std > 0 else 0.0
    
    # 信号：z_score > 1.5 → 做空；z_score < -1.5 → 做多
    signal = -z_score  # 反向因子
    
    return {
        'z_score': z_score,
        'signal': signal,
        'raw_rate': current,
    }
```

|------|----|------|

| 7d | 0.18 | **强**（OI 变化领先价格） |

```python
# fetch_oi_factor.py
def calculate_oi_factor(symbol, lookback=60):
    """计算 Open Interest 因子"""
    oi = fetch_open_interest(symbol, limit=lookback)
    
    # 特征 1：OI 是否创新高
    oi_new_high = 1.0 if oi[-1] == max(oi) else 0.0
    
    # 特征 2：OI 7 日变化率
    oi_change_7d = (oi[-1] - oi[-7]) / oi[-7]
    
    # 特征 3：OI/Price 背离（OI↑ 但 Price↓ → 看跌）

    price_change_7d = (prices[-1] - prices[-7]) / prices[-7]
    
    divergence = 1.0 if (oi_change_7d > 0.05 and price_change_7d < 0) else 0.0
    
    signal = oi_new_high * 0.3 + oi_change_7d * 0.5 - divergence * 0.2
    
    return {
        'oi_new_high': oi_new_high,
        'oi_change_7d': oi_change_7d,
        'divergence': divergence,
        'signal': signal,
    }
```

## 7. 因子 IC 分析与优化

### IC（Information Coefficient）
```
定义：因子值与未来收益的相关系数
计算：pearsonr(factor_values, future_returns)
```

### 因子评估框架
```python
# factor_ic_analysis.py
import numpy as np
from scipy.stats import pearsonr

def calculate_ic(factor_values, future_returns):
    """
    计算 IC（Information Coefficient）
    """
    # 去除 NaN
    mask = ~(np.isnan(factor_values) | np.isnan(future_returns))
    factor_values = factor_values[mask]
    future_returns = future_returns[mask]
    
    if len(factor_values) < 10:
        return np.nan
    
    ic, p_value = pearsonr(factor_values, future_returns)
    
    return ic

def factor_ic_test(symbol, factor_name, lookback=252):
    """
    因子 IC 测试（252 个交易日）
    """
    # 获取因子值（每日）
    factor_values = fetch_factor_history(symbol, factor_name, lookback)
    
    # 获取未来 7 日收益
    future_returns = calculate_future_returns(symbol, lookback, horizon=7)
    
    # 计算 IC
    ic = calculate_ic(factor_values, future_returns)
    
    # IC_IR（信息比率）= IC / IC_std
    ic_series = []
    for i in range(lookback - 20):
        ic_i = calculate_ic(factor_values[i:i+20], future_returns[i:i+20])
        ic_series.append(ic_i)
    
    ic_ir = np.mean(ic_series) / np.std(ic_series) if np.std(ic_series) > 0 else 0.0
    
    return {
        'factor': factor_name,
        'ic': ic,
        'ic_ir': ic_ir,
        'rating': 'STRONG' if abs(ic) > 0.15 else 'MEDIUM' if abs(ic) > 0.08 else 'WEAK'
    }

# 运行所有因子的 IC 测试
if __name__ == '__main__':
    factors = ['funding_rate', 'open_interest', 'twitter_hype', 'stablecoin_inflow']
    
    for factor in factors:
        result = factor_ic_test('BTCUSDT', factor)
        print(f"{factor:20} IC={result['ic']:.3f}  IC_IR={result['ic_ir']:.2f}  {result['rating']}")
```

```python
def equal_weight_combo(factor_signals):
```

### 方法 2：IC IR 加权
```python
def ic_ir_weighted_combo(factor_signals, factor_ic_irs):
    """按 IC_IR 加权"""
    weights = np.array([factor_ic_irs[f] for f in factor_signals])
    weights = weights / np.sum(np.abs(weights))  # 归一化
    
    signals = np.array([factor_signals[f] for f in factor_signals])
    
    return np.dot(weights, signals)
```

### 方法 3：机器学习加权（LightGBM）
```python
# factor_ml_combo.py
import lightgbm as lgb

def ml_factor_combo(factor_data, returns, test_size=0.3):
    """
    用 LightGBM 学习因子加权
    """
    # 训练测试分割
    split_idx = int(len(returns) * (1 - test_size))
    
    X_train = factor_data[:split_idx]
    y_train = returns[:split_idx]
    
    X_test = factor_data[split_idx:]
    y_test = returns[split_idx:]
    
    # 训练模型
    model = lgb.LGBMRegressor(num_leaves=31, learning_rate=0.05, n_estimators=100)
    model.fit(X_train, y_train)
    
    # 预测
    y_pred = model.predict(X_test)
    
    # 计算组合收益
    portfolio_returns = np.sum(y_pred * X_test, axis=1)
    
    sharpe = np.mean(portfolio_returns) / np.std(portfolio_returns) * np.sqrt(252)
    
    return {'model': model, 'sharpe': sharpe}
```

```python
import sys

def multi_factor_strategy(symbol, candles, factors=['funding', 'oi', 'twitter'], weights=None):
    """
    多因子策略
    """
    # 计算所有因子
    factor_values = {}
    for factor in factors:
        factor_values[factor] = calc_factor(factor, symbol, candles)
    
    # 归一化到 [0, 1]
    normalized = {}
    for k, v in factor_values.items():
        v = np.array(v)
        v = (v - np.min(v)) / (np.max(v) - np.min(v) + 1e-8)
        normalized[k] = v
    
    # 加权合成信号
    if weights is None:
        weights = {f: 1.0/len(factors) for f in factors}  # 等权
    
    signals = [None] * len(candles)
    for i in range(len(candles)):
        score = sum(normalized[f][i] * weights[f] for f in factors)
        
        if score > 0.6:
            signals[i] = 'BUY'
        elif score < 0.4:
            signals[i] = 'SELL'
    
    return signals
```

```python
import pandas as pd
import numpy as np

class MovingAverages:
    """移动平均线系列"""
    
    @staticmethod
    def sma(prices: list, period: int) -> list:
        """
        简单移动平均 (SMA)
        SMA = (P1 + P2 + ... + Pn) / n
        """
        prices = np.array(prices)
        return pd.Series(prices).rolling(window=period).mean().tolist()
    
    @staticmethod
    def ema(prices: list, period: int) -> list:
        """
        指数移动平均 (EMA)
        EMA = α * Price + (1 - α) * EMA_prev
        α = 2 / (n + 1)
        """
        prices = np.array(prices)
        return pd.Series(prices).ewm(span=period, adjust=False).mean().tolist()
    
    @staticmethod
    def wma(prices: list, period: int) -> list:
        """
        加权移动平均 (WMA)
        近期价格权重更大
        """
        prices = np.array(prices)
        weights = np.arange(1, period + 1)
        return pd.Series(prices).rolling(period).apply(
            lambda x: np.dot(x, weights) / weights.sum(), raw=True
        ).tolist()
    
    @staticmethod
    def dema(prices: list, period: int) -> list:
        """
        双指数移动平均 (DEMA)
        减少 EMA 的滞后性
        """
        prices = np.array(prices)
        ema1 = pd.Series(prices).ewm(span=period, adjust=False).mean()
        ema2 = ema1.ewm(span=period, adjust=False).mean()
        return (2 * ema1 - ema2).tolist()
    
    @staticmethod
    def tema(prices: list, period: int) -> list:
        """
        三指数移动平均 (TEMA)
        进一步减少滞后
        """
        prices = np.array(prices)
        ema1 = pd.Series(prices).ewm(span=period, adjust=False).mean()
        ema2 = ema1.ewm(span=period, adjust=False).mean()
        ema3 = ema2.ewm(span=period, adjust=False).mean()
        return (3 * ema1 - 3 * ema2 + ema3).tolist()
    
    @staticmethod
    def hma(prices: list, period: int) -> list:
        """
        Hull 移动平均 (HMA)
        低滞后、高平滑
        HMA = WMA(2*WMA(n/2) - WMA(n), sqrt(n))
        """
        prices = np.array(prices)
        half_period = period // 2
        sqrt_period = int(np.sqrt(period))
        
        wma_half = pd.Series(prices).rolling(half_period).apply(
            lambda x: np.dot(x, np.arange(1, half_period + 1)) / sum(range(1, half_period + 1)), raw=True
        )
        wma_full = pd.Series(prices).rolling(period).apply(
            lambda x: np.dot(x, np.arange(1, period + 1)) / sum(range(1, period + 1)), raw=True
        )
        raw_hma = 2 * wma_half - wma_full
        
        return raw_hma.rolling(sqrt_period).apply(
            lambda x: np.dot(x, np.arange(1, sqrt_period + 1)) / sum(range(1, sqrt_period + 1)), raw=True
        ).tolist()


class TrendIndicators:
    """趋势指标"""
    
    @staticmethod
    def adx(highs: list, lows: list, closes: list, period: int = 14) -> dict:
        """
        平均趋向指数 (ADX)
        衡量趋势强度（0-100）
        ADX > 25 → 趋势强
        ADX < 20 → 无趋势
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        # True Range
        tr1 = highs - lows
        tr2 = np.abs(highs - np.roll(closes, 1))
        tr3 = np.abs(lows - np.roll(closes, 1))
        tr = np.maximum(tr1, np.maximum(tr2, tr3))
        
        # Directional Movement
        plus_dm = np.where(
            (highs - np.roll(highs, 1)) > (np.roll(lows, 1) - lows),
            np.maximum(highs - np.roll(highs, 1), 0), 0
        )
        minus_dm = np.where(
            (np.roll(lows, 1) - lows) > (highs - np.roll(highs, 1)),
            np.maximum(np.roll(lows, 1) - lows, 0), 0
        )
        
        # Smooth
        atr = pd.Series(tr).rolling(period).mean()
        plus_di = 100 * pd.Series(plus_dm).rolling(period).mean() / atr
        minus_di = 100 * pd.Series(minus_dm).rolling(period).mean() / atr
        
        # DX and ADX
        dx = 100 * np.abs(plus_di - minus_di) / (plus_di + minus_di)
        adx = pd.Series(dx).rolling(period).mean()
        
        return {
            "adx": adx.tolist(),
            "plus_di": plus_di.tolist(),
            "minus_di": minus_di.tolist(),
            "signal": ["强趋势" if x > 25 else "弱趋势" if x > 20 else "震荡" for x in adx]
        }
    
    @staticmethod
    def supertrend(highs: list, lows: list, closes: list, period: int = 10, multiplier: float = 3.0) -> dict:
        """
        Supertrend 指标
        结合 ATR 和价格过滤的趋势跟踪系统
        
        优点：
        - 滞后性低
        - 明确多空信号
        - 止损清晰
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        # ATR
        tr1 = highs - lows
        tr2 = np.abs(highs - np.roll(closes, 1))
        tr3 = np.abs(lows - np.roll(closes, 1))
        tr = np.maximum(tr1, np.maximum(tr2, tr3))
        atr = pd.Series(tr).rolling(period).mean()
        
        # Basic Upper/Lower Band
        hl2 = (highs + lows) / 2
        upper_band = hl2 + multiplier * atr
        lower_band = hl2 - multiplier * atr
        
        # Supertrend
        supertrend = np.zeros_like(closes)
        direction = np.ones_like(closes)  # 1 = 多头, -1 = 空头
        
        for i in range(period, len(closes)):
            if closes[i] > upper_band[i-1]:
                direction[i] = 1
            elif closes[i] < lower_band[i-1]:
                direction[i] = -1
            else:
                direction[i] = direction[i-1]
            
            upper_band[i] = min(upper_band[i], upper_band[i-1]) if direction[i] == -1 else upper_band[i]
            lower_band[i] = max(lower_band[i], lower_band[i-1]) if direction[i] == 1 else lower_band[i]
            
            supertrend[i] = upper_band[i] if direction[i] == -1 else lower_band[i]
        
        return {
            "supertrend": supertrend.tolist(),
            "direction": direction.tolist(),
            "upper_band": upper_band.tolist(),
            "lower_band": lower_band.tolist(),
            "signal": ["做多" if d == 1 else "做空" for d in direction]
        }
    
    @staticmethod
    def ichimoku(highs: list, lows: list, closes: list) -> dict:
        """
        Ichimoku Kinko Hyo (一目均衡表)
        
        组成部分：
        1. Tenkan-sen（转折线）：(9日高+9日低)/2
        2. Kijun-sen（基准线）：(26日高+26日低)/2
        3. Senkou Span A（先行带A）：(转折线+基准线)/2，向前26日
        4. Senkou Span B（先行带B）：(52日高+52日低)/2，向前26日
        5. Chikou Span（延迟线）：当前收盘，向后26日
        6. 云带：先行带A与先行带B之间的区域
        """
        highs = np.array(highs)
        lows = np.array(lows)
        
        def ichimoku_line(period):
            return (pd.Series(highs).rolling(period).max() + pd.Series(lows).rolling(period).min()) / 2
        
        tenkan = ichimoku_line(9)
        kijun = ichimoku_line(26)
        senkou_a = ((tenkan + kijun) / 2).shift(26)
        senkou_b = (ichimoku_line(52) / 2).shift(26)
        chikou = pd.Series(closes).shift(-26)
        
        return {
            "tenkan_sen": tenkan.tolist(),
            "kijun_sen": kijun.tolist(),
            "senkou_span_a": senkou_a.tolist(),
            "senkou_span_b": senkou_b.tolist(),
            "chikou_span": chikou.tolist(),
            "cloud_top": np.maximum(senkou_a, senkou_b).tolist(),
            "cloud_bottom": np.minimum(senkou_a, senkou_b).tolist()
        }
    
    @staticmethod
    def parabolicsar(highs: list, lows: list, closes: list, start: float = 0.02, increment: float = 0.02, max_step: float = 0.2) -> list:
        """
        Parabolic SAR（抛物线止损）
        
        参数：
        - start: 初始 AF
        - increment: 每当 EP 更新时 AF 增加量
        - max_step: AF 最大值
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        sar = np.zeros_like(closes)
        trend = np.ones_like(closes)  # 1 = 多头, -1 = 空头
        af = start
        ep = highs[0]
        
        sar[0] = lows[0]
        
        for i in range(1, len(closes)):
            prev_sar = sar[i-1]
            
            if trend[i-1] == 1:
                sar[i] = prev_sar + af * (ep - prev_sar)
                if lows[i] < sar[i]:
                    trend[i] = -1
                    sar[i] = ep
                    af = start
                    ep = lows[i]
                else:
                    trend[i] = 1
                    if highs[i] > ep:
                        ep = highs[i]
                        af = min(af + increment, max_step)
            else:
                sar[i] = prev_sar - af * (prev_sar - ep)
                if highs[i] > sar[i]:
                    trend[i] = 1
                    sar[i] = ep
                    af = start
                    ep = highs[i]
                else:
                    trend[i] = -1
                    if lows[i] < ep:
                        ep = lows[i]
                        af = min(af + increment, max_step)
        
        return {
            "sar": sar.tolist(),
            "trend": trend.tolist(),
            "signal": ["做多" if t == 1 else "做空" for t in trend]
        }
```

```python
class MomentumIndicators:
    """动量指标"""
    
    @staticmethod
    def rsi(prices: list, period: int = 14) -> dict:
        """
        相对强弱指数 (RSI)
        
        解读：
        - RSI > 70 → 超买
        - RSI < 30 → 超卖
        - RSI > 50 → 偏多
        - RSI < 50 → 偏空
        """
        prices = np.array(prices)
        deltas = np.diff(prices, prepend=prices[0])
        
        gains = np.where(deltas > 0, deltas, 0)
        losses = np.where(deltas < 0, -deltas, 0)
        
        avg_gain = pd.Series(gains).rolling(period).mean()
        avg_loss = pd.Series(losses).rolling(period).mean()
        
        rs = avg_gain / avg_loss
        rsi = 100 - (100 / (1 + rs))
        
        # RSI 信号
        signals = []
        for r in rsi:
            if pd.isna(r):
                signals.append(None)
            elif r > 70:
                signals.append("超买")
            elif r < 30:
                signals.append("超卖")
            elif r > 50:
                signals.append("偏多")
            else:
                signals.append("偏空")
        
        return {
            "rsi": rsi.tolist(),
            "signal": signals,
            "avg_gain": avg_gain.tolist(),
            "avg_loss": avg_loss.tolist()
        }
    
    @staticmethod
    def stochastic(highs: list, lows: list, closes: list, k_period: int = 14, d_period: int = 3) -> dict:
        """
        随机指标 (Stochastic Oscillator)
        
        %K：当前收盘在N日内区间的位置
        %D：%K的SMA
        
        解读：
        - %K > 80 → 超买
        - %K < 20 → 超卖
        - %K 上穿 %D → 买入信号
        - %K 下穿 %D → 卖出信号
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        lowest_low = pd.Series(lows).rolling(k_period).min()
        highest_high = pd.Series(highs).rolling(k_period).max()
        
        k = 100 * (closes - lowest_low) / (highest_high - lowest_low)
        d = k.rolling(d_period).mean()
        
        # 金叉/死叉信号
        signals = []
        for i in range(len(k)):
            if pd.isna(k[i]) or pd.isna(d[i]):
                signals.append(None)
            elif i > 0 and not pd.isna(k[i-1]) and not pd.isna(d[i-1]):
                if k[i] > d[i] and k[i-1] <= d[i-1]:
                    signals.append("金叉买入")
                elif k[i] < d[i] and k[i-1] >= d[i-1]:
                    signals.append("死叉卖出")
                else:
                    signals.append("持仓")
            else:
                signals.append(None)
        
        return {
            "k": k.tolist(),
            "d": d.tolist(),
            "signal": signals,
            "overbought": k > 80,
            "oversold": k < 20
        }
    
    @staticmethod
    def macd(prices: list, fast: int = 12, slow: int = 26, signal: int = 9) -> dict:
        """
        MACD (Moving Average Convergence Divergence)
        
        MACD Line = EMA(12) - EMA(26)
        Signal Line = EMA(9) of MACD
        Histogram = MACD - Signal
        
        解读：
        - MACD > 0 → 多头
        - MACD < 0 → 空头
        - MACD 上穿 Signal → 买入
        - MACD 下穿 Signal → 卖出
        - 价格创新高但 MACD 没创新高 → 顶背离
        - 价格创新低但 MACD 没创新低 → 底背离
        """
        prices = np.array(prices)
        
        ema_fast = pd.Series(prices).ewm(span=fast, adjust=False).mean()
        ema_slow = pd.Series(prices).ewm(span=slow, adjust=False).mean()
        
        macd_line = ema_fast - ema_slow
        signal_line = macd_line.ewm(span=signal, adjust=False).mean()
        histogram = macd_line - signal_line
        
        # 交叉信号
        cross_signals = []
        for i in range(len(macd_line)):
            if pd.isna(macd_line[i]) or pd.isna(signal_line[i]):
                cross_signals.append(None)
            elif i > 0 and not pd.isna(macd_line[i-1]) and not pd.isna(signal_line[i-1]):
                if macd_line[i] > signal_line[i] and macd_line[i-1] <= signal_line[i-1]:
                    cross_signals.append("MACD 金叉")
                elif macd_line[i] < signal_line[i] and macd_line[i-1] >= signal_line[i-1]:
                    cross_signals.append("MACD 死叉")
                else:
                    cross_signals.append(None)
            else:
                cross_signals.append(None)
        
        return {
            "macd": macd_line.tolist(),
            "signal": signal_line.tolist(),
            "histogram": histogram.tolist(),
            "cross_signal": cross_signals,
            "trend": ["多头" if m > s else "空头" if m < s else "中性" 
                       for m, s in zip(macd_line, signal_line)]
        }
    
    @staticmethod
    def momentum(prices: list, period: int = 10) -> list:
        """
        动量指标 (Momentum)
        当前价格 - N日前价格
        """
        prices = np.array(prices)
        return (prices - np.roll(prices, period)).tolist()
    
    @staticmethod
    def roc(prices: list, period: int = 10) -> list:
        """
        价格变化率 (Rate of Change)
        ROC = (Price - Price_n) / Price_n * 100
        """
        prices = np.array(prices)
        return 100 * (prices - np.roll(prices, period)) / np.roll(prices, period)
    
    @staticmethod
    def cci(highs: list, lows: list, closes: list, period: int = 20) -> dict:
        """
        商品通道指数 (CCI)
        
        解读：
        - CCI > 100 → 超买
        - CCI < -100 → 超卖
        - CCI 从负值上穿 -100 → 买入
        - CCI 从正值下穿 100 → 卖出
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        tp = (highs + lows + closes) / 3
        sma_tp = pd.Series(tp).rolling(period).mean()
        mad = pd.Series(np.abs(tp - sma_tp)).rolling(period).mean()
        
        cci = (tp - sma_tp) / (0.015 * mad)
        
        return {
            "cci": cci.tolist(),
            "signal": ["超买" if x > 100 else "超卖" if x < -100 else "中性" for x in cci]
        }
```

```python
class VolatilityIndicators:
    """波动率指标"""
    
    @staticmethod
    def atr(highs: list, lows: list, closes: list, period: int = 14) -> list:
        """
        平均真实范围 (ATR)
        衡量市场波动性
        
        ATR 常用于：
        - 设置止损（1.5-2x ATR）
        - 波动率归一化
        - 趋势强度判断
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        tr1 = highs - lows
        tr2 = np.abs(highs - np.roll(closes, 1))
        tr3 = np.abs(lows - np.roll(closes, 1))
        tr = np.maximum(tr1, np.maximum(tr2, tr3))
        
        return pd.Series(tr).ewm(span=period, adjust=True).mean().tolist()
    
    @staticmethod
    def keltner_channels(highs: list, lows: list, closes: list, ema_period: int = 20, atr_period: int = 10, multiplier: float = 2.0) -> dict:
        """
        Keltner 通道
        基于 ATR 的趋势通道
        
        中轨：EMA
        上轨：EMA + 2 * ATR
        下轨：EMA - 2 * ATR
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        
        ema = pd.Series(closes).ewm(span=ema_period, adjust=False).mean()
        atr = VolatilityIndicators.atr(highs, lows, closes, atr_period)
        
        upper = ema + multiplier * pd.Series(atr)
        lower = ema - multiplier * pd.Series(atr)
        
        # 价格位置
        price_position = (closes - lower) / (upper - lower)
        
        return {
            "middle": ema.tolist(),
            "upper": upper.tolist(),
            "lower": lower.tolist(),
            "atr": atr,
            "bandwidth": ((upper - lower) / ema * 100).tolist(),
            "price_position": price_position.tolist()
        }
    
    @staticmethod
    def bollinger_bands(prices: list, period: int = 20, std_dev: float = 2.0) -> dict:
        """
        布林带
        中轨：SMA
        上轨：SMA + 2 * STD
        下轨：SMA - 2 * STD
        
        布林带收缩 → 突破在即
        价格触碰下轨 → 超卖
        价格触碰上轨 → 超买
        """
        prices = np.array(prices)
        
        sma = pd.Series(prices).rolling(period).mean()
        std = pd.Series(prices).rolling(period).std()
        
        upper = sma + std_dev * std
        lower = sma - std_dev * std
        
        # %B 指标（价格在布林带的位置）
        pct_b = (prices - lower) / (std_dev * std * 2)
        
        # 带宽（布林带宽度，反映波动率）
        bandwidth = (upper - lower) / sma * 100
        
        return {
            "middle": sma.tolist(),
            "upper": upper.tolist(),
            "lower": lower.tolist(),
            "pct_b": pct_b.tolist(),
            "bandwidth": bandwidth.tolist(),
            "signal": ["超买" if pb > 1 else "超卖" if pb < 0 else "中性" for pb in pct_b]
        }
    
    @staticmethod
    def donchian_channels(highs: list, lows: list, period: int = 20) -> dict:
        """
        唐奇安通道
        上轨：N日最高价
        下轨：N日最低价
        
        价格突破上轨 → 多头信号
        价格跌破下轨 → 空头信号
        """
        highs = np.array(highs)
        lows = np.array(lows)
        
        upper = pd.Series(highs).rolling(period).max()
        lower = pd.Series(lows).rolling(period).min()
        middle = (upper + lower) / 2
        
        return {
            "upper": upper.tolist(),
            "middle": middle.tolist(),
            "lower": lower.tolist()
        }
```

```python
    def obv(closes: list, volumes: list) -> list:
        """
        能量潮 (OBV)
        累计成交量（上涨日记加，下跌日记减）
        
        OBV 上升 → 资金流入
        OBV 下降 → 资金流出
        OBV 创新高 → 确认上涨趋势
        """
        closes = np.array(closes)
        volumes = np.array(volumes)
        
        direction = np.sign(np.diff(closes, prepend=closes[0]))
        obv = np.cumsum(direction * volumes)
        
        return obv.tolist()
    
    @staticmethod
    def vwap(highs: list, lows: list, closes: list, volumes: list) -> list:
        """
        成交量加权平均价 (VWAP)
        
        用途：
        - 日内交易基准
        - VWAP 以上 → 偏多
        - VWAP 以下 → 偏空
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        volumes = np.array(volumes)
        
        typical_price = (highs + lows + closes) / 3
        cumulative_tp_vol = np.cumsum(typical_price * volumes)
        cumulative_vol = np.cumsum(volumes)
        
        vwap = cumulative_tp_vol / cumulative_vol
        
        return vwap.tolist()
    
    @staticmethod
    def mfi(highs: list, lows: list, closes: list, volumes: list, period: int = 14) -> dict:
        """
        资金流量指数 (MFI)
        类似于 RSI，但使用成交量
        
        MFI > 80 → 超买
        MFI < 20 → 超卖
        MFI 与价格背离 → 反转信号
        """
        highs = np.array(highs)
        lows = np.array(lows)
        closes = np.array(closes)
        volumes = np.array(volumes)
        
        typical_price = (highs + lows + closes) / 3
        raw_money_flow = typical_price * volumes
        
        positive_flow = np.where(np.diff(typical_price, prepend=typical_price[0]) > 0, raw_money_flow, 0)
        negative_flow = np.where(np.diff(typical_price, prepend=typical_price[0]) < 0, raw_money_flow, 0)
        
        positive_mf = pd.Series(positive_flow).rolling(period).sum()
        negative_mf = pd.Series(negative_flow).rolling(period).sum()
        
        mf_ratio = positive_mf / negative_mf
        mfi = 100 - (100 / (1 + mf_ratio))
        
        return {
            "mfi": mfi.tolist(),
            "signal": ["超买" if x > 80 else "超卖" if x < 20 else "中性" for x in mfi]
        }
    
    @staticmethod
    def vwap_ma_cross(closes: list, volumes: list, highs: list, lows: list, short_ma: int = 5, long_ma: int = 20) -> dict:
        """
        VWAP 均线交叉策略
        短期 VWAP 上穿长期 VWAP → 买入
        短期 VWAP 下穿长期 VWAP → 卖出
        """
        # 计算 VWAP（简化，使用每日）
        vwap = VolumeIndicators.vwap(highs, lows, closes, volumes)
        
        # 计算 VWAP 的 MA
        vwap_ma_short = pd.Series(vwap).rolling(short_ma).mean()
        vwap_ma_long = pd.Series(vwap).rolling(long_ma).mean()
        
        # 交叉信号
        signals = []
        for i in range(len(vwap_ma_short)):
            if pd.isna(vwap_ma_short[i]) or pd.isna(vwap_ma_long[i]):
                signals.append(None)
            elif i > 0 and not pd.isna(vwap_ma_short[i-1]):
                if vwap_ma_short[i] > vwap_ma_long[i] and vwap_ma_short[i-1] <= vwap_ma_long[i-1]:
                    signals.append("买入")
                elif vwap_ma_short[i] < vwap_ma_long[i] and vwap_ma_short[i-1] >= vwap_ma_long[i-1]:
                    signals.append("卖出")
                else:
                    signals.append("持仓")
            else:
                signals.append(None)
        
        return {
            "vwap": vwap,
            "vwap_ma_short": vwap_ma_short.tolist(),
            "vwap_ma_long": vwap_ma_long.tolist(),
            "signal": signals
        }
```

```python
class MAStrategy:
    """均线系统策略"""
    
    @staticmethod
    def golden_death_cross(short_prices: list, long_prices: list) -> dict:
        """
        黄金交叉/死亡交叉
        
        黄金交叉：短期 MA 上穿长期 MA → 买入信号
        死亡交叉：短期 MA 下穿长期 MA → 卖出信号
        """
        short_ma = pd.Series(short_prices).rolling(50).mean()
        long_ma = pd.Series(long_prices).rolling(200).mean()
        
        signals = []
        for i in range(len(short_ma)):
            if pd.isna(short_ma[i]) or pd.isna(long_ma[i]):
                signals.append(None)
            elif i > 0 and not pd.isna(short_ma[i-1]):
                if short_ma.iloc[i] > long_ma.iloc[i] and short_ma.iloc[i-1] <= long_ma.iloc[i-1]:
                    signals.append("黄金交叉 → 买入")
                elif short_ma.iloc[i] < long_ma.iloc[i] and short_ma.iloc[i-1] >= long_ma.iloc[i-1]:
                    signals.append("死亡交叉 → 卖出")
                else:
                    signals.append(None)
            else:
                signals.append(None)
        
        return {
            "short_ma_50": short_ma.tolist(),
            "long_ma_200": long_ma.tolist(),
            "signal": signals
        }
    
    @staticmethod
    def triple_ma(prices: list, ma1: int = 5, ma2: int = 20, ma3: int = 60) -> dict:
        """
        三均线系统
        
        多头排列：MA1 > MA2 > MA3 → 强上涨趋势
        空头排列：MA1 < MA2 < MA3 → 强下跌趋势
        混乱排列：趋势不明
        """
        prices = np.array(prices)
        
        ma_1 = pd.Series(prices).rolling(ma1).mean()
        ma_2 = pd.Series(prices).rolling(ma2).mean()
        ma_3 = pd.Series(prices).rolling(ma3).mean()
        
        alignment = []
        for i in range(len(prices)):
            if pd.isna(ma_3.iloc[i]):
                alignment.append(None)
            elif ma_1.iloc[i] > ma_2.iloc[i] > ma_3.iloc[i]:
                alignment.append("多头排列")
            elif ma_1.iloc[i] < ma_2.iloc[i] < ma_3.iloc[i]:
                alignment.append("空头排列")
            else:
                alignment.append("混乱")
        
        return {
            f"ma_{ma1}": ma_1.tolist(),
            f"ma_{ma2}": ma_2.tolist(),
            f"ma_{ma3}": ma_3.tolist(),
            "alignment": alignment
        }
```

```python
class Web3Indicators:
    """Web3 专属量化指标"""
    
    @staticmethod
    def funding_rate_anomaly(funding_rates: list, threshold: float = 0.1) -> dict:
        """
        资金费率异常检测
        
        极端正值 → 市场过热，多头被挤压风险
        极端负值 → 市场恐慌，空头被挤压风险
        """
        funding_rates = np.array(funding_rates)
        
        ma = pd.Series(funding_rates).rolling(7).mean()
        std = pd.Series(funding_rates).rolling(7).std()
        
        z_score = (funding_rates - ma) / std
        
        return {
            "funding_rate": funding_rates.tolist(),
            "ma_7d": ma.tolist(),
            "z_score": z_score.tolist(),
            "anomaly": np.abs(z_score) > 2,
            "extreme_positive": funding_rates > threshold,
            "extreme_negative": funding_rates < -threshold
        }
    
    @staticmethod
    def whale_wallet_divergence(
        whale_addresses: list,
        price_history: list,
        threshold_pct: float = 0.1
    ) -> dict:
        """
        鲸鱼钱包异动检测
        
        检测鲸鱼地址的总持仓变化是否与价格走势背离
        """
        whale_holdings = [get_wallet_balance(addr) for addr in whale_addresses]
        whale_total_change = np.diff(whale_holdings, prepend=whale_holdings[0]) / whale_holdings[0]
        price_change = np.diff(price_history, prepend=price_history[0]) / price_history[0]
        
        # 背离检测
        divergence = whale_total_change - price_change
        
        return {
            "whale_total_change_pct": whale_total_change.tolist(),
            "price_change_pct": price_change.tolist(),
            "divergence": divergence.tolist(),
            "bullish_divergence": (whale_total_change > threshold_pct) & (price_change < -threshold_pct),
            "bearish_divergence": (whale_total_change < -threshold_pct) & (price_change > threshold_pct)
        }
    
    @staticmethod
    def gas_volatility(gas_prices_gwei: list, period: int = 7) -> dict:
        """
        Gas 波动率分析
        
        Gas 突然飙升 → DeFi 活动增加（可能是大动作）
        Gas 极低 → 市场冷清
        """
        gas = np.array(gas_prices_gwei)
        
        ma = pd.Series(gas).rolling(period).mean()
        std = pd.Series(gas).rolling(period).std()
        
        cv = std / ma  # 变异系数
        
        return {
            "gas_gwei": gas.tolist(),
            "ma": ma.tolist(),
            "std": std.tolist(),
            "cv": cv.tolist(),
            "signal": ["Gas 飙升" if g > ma.iloc[i] * 2 else "Gas 正常" 
                       for i, g in enumerate(gas)]
        }
```

```python
def multi_indicator_strategy(
    prices: list,
    highs: list,
    lows: list,
    volumes: list
) -> dict:
    """
    多指标共振策略
    
    买入信号：至少 4 个指标看多
    卖出信号：至少 4 个指标看空
    """
    # 计算各指标
    rsi = MomentumIndicators.rsi(prices, 14)
    macd = MomentumIndicators.macd(prices)
    bb = VolatilityIndicators.bollinger_bands(prices)
    supertrend = TrendIndicators.supertrend(highs, lows, prices)
    obv = VolumeIndicators.obv(prices, volumes)
    
    signals = []
    for i in range(len(prices)):
        bullish_count = 0
        bearish_count = 0
        
        # RSI
        if not pd.isna(rsi["rsi"][i]):
            if rsi["rsi"][i] < 30: bullish_count += 1
            elif rsi["rsi"][i] > 70: bearish_count += 1
        
        # MACD
        if not pd.isna(macd["macd"][i]):
            if macd["macd"][i] > macd["signal"][i]: bullish_count += 1
            else: bearish_count += 1
        
        # Bollinger Bands
        if not pd.isna(bb["pct_b"][i]):
            if bb["pct_b"][i] < 0: bullish_count += 1
            elif bb["pct_b"][i] > 1: bearish_count += 1
        
        # Supertrend
        if not pd.isna(supertrend["direction"][i]):
            if supertrend["direction"][i] == 1: bullish_count += 1
            else: bearish_count += 1
        
        # OBV Trend
        if i > 10 and not pd.isna(obv[i]):
            if obv[i] > obv[i-5]: bullish_count += 1
            else: bearish_count += 1
        
        total = bullish_count + bearish_count
        if total >= 4:
            if bullish_count >= 4:
                signals.append("强买入")
            elif bearish_count >= 4:
                signals.append("强卖出")
            else:
                signals.append("中性")
        else:
            signals.append("中性")
    
    return {
        "signal": signals,
        "confidence": [
            {"bullish": signals.count("强买入") / len(signals) * 100}
        ]
    }
```
