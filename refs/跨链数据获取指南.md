```python
def analyze_cross_chain_flows(days=7) -> dict:
    """
    分析跨链资金流向
    用于判断市场情绪和资金轮动
    """
    flows = {}
    
    # 从 Dune 或 DeFiLlama 获取数据
    for bridge in ["arbitrum", "optimism", "hop", "stargate"]:
        flows[bridge] = {
            "deposits": fetch_bridge_deposits(bridge, days),
            "withdrawals": fetch_bridge_withdrawals(bridge, days),
            "net_flow": calculate_net_flow(bridge, days)
        }
    
    # 计算总体流向
    total_deposits = sum(f["deposits"] for f in flows.values())
    total_withdrawals = sum(f["withdrawals"] for f in flows.values())
    
    # 净流向
    # 正数 = 资金流入 L2（看涨）
    # 负数 = 资金回流 L1（看跌）
    net_flow = total_deposits - total_withdrawals
    
    return {
        "total_deposits_7d": total_deposits,
        "total_withdrawals_7d": total_withdrawals,
        "net_flow_7d": net_flow,
        "net_flow_direction": "L2_Inflow" if net_flow > 0 else "L1_Inflow",
        "bridge_flows": flows,
        "sentiment": "bullish" if net_flow > 100_000_000 else "neutral" if net_flow > 0 else "bearish"
    }

def detect_capital_rotation() -> dict:
    """
    检测资金轮动
    资金通常在 L1 -> L2 或 L2 -> L1 之间轮动
    """
    # 获取各链 TVL 变化
    tvl_changes = {
        "ethereum": get_tvl_change("ethereum", days=7),
        "arbitrum": get_tvl_change("arbitrum", days=7),
        "optimism": get_tvl_change("optimism", days=7),
        "base": get_tvl_change("base", days=7)
    }
    
    # 找出 TVL 增长最多的链
    sorted_chains = sorted(tvl_changes.items(), key=lambda x: x[1], reverse=True)
    
    # 轮动信号
    rotation_signals = []
    
    for i, (chain, change) in enumerate(sorted_chains[:2]):
        if change > 10:  # 增长 > 10%
            rotation_signals.append({
                "chain": chain,
                "tvl_change_pct": change,
                "rotation_type": "inflow",
                "confidence": "HIGH" if change > 20 else "MEDIUM"
            })
        elif change < -10:  # 下降 > 10%
            rotation_signals.append({
                "chain": chain,
                "tvl_change_pct": change,
                "rotation_type": "outflow",
                "confidence": "HIGH" if change < -20 else "MEDIUM"
            })
    
    return {
        "tvl_changes": tvl_changes,
        "rotation_signals": rotation_signals,
        "leading_chain": sorted_chains[0][0] if sorted_chains else None,
        "strategy_implication": get_rotation_strategy(sorted_chains)
    }
```

```python
class SolanaData:
    """Solana 链上数据获取"""
    
    def __init__(self, rpc_url="https://api.mainnet-beta.solana.com"):
        self.rpc = rpc_url
        self.jito_rpc = "https://arb组-mainnet.jito.wtf/api/v1"  # Jito RPC
    
    def get_account_info(self, account: str) -> dict:
        """获取账户信息"""
        payload = {
            "jsonrpc": "2.0",
            "id": 1,
            "method": "getAccountInfo",
            "params": [account, {"encoding": "jsonParsed"}]
        }
        
        response = requests.post(self.rpc, json=payload)
        return response.json()
    
    def get_token_accounts(self, owner: str) -> list:
        """获取钱包的所有 Token 账户"""
        payload = {
            "jsonrpc": "2.0",
            "id": 1,
            "method": "getTokenAccountsByOwner",
            "params": [
                owner,
                {"mint": "TokenkegQfeZyiNwAJbNbGKPFXCWuBvf9Ss623VQ5DA"},  # USDC Mint
                {"encoding": "jsonParsed"}
            ]
        }
        
        response = requests.post(self.rpc, json=payload)
        return response.json().get("result", {}).get("value", [])
    
    def get_swap_quote(self, input_mint: str, output_mint: str, amount: int) -> dict:
        """
        获取 Jupiter 报价（Solana DEX 聚合器）
        """
        # Jupiter API
        url = f"https://quote-api.jup.ag/v6/quote"
        params = {
            "inputMint": input_mint,
            "outputMint": output_mint,
            "amount": amount,
            "slippageBps": 50  # 0.5% 滑点
        }
        
        response = requests.get(url, params=params)
        return response.json()
    
    def get_market_depth(self, market_address: str, depth=20) -> dict:
        """
        获取订单簿深度（Raydium / Serum / Phoenix）
        """
        # 使用 OpenBook/Coral 订单簿
        payload = {
            "jsonrpc": "2.0",
            "id": 1,
            "method": "getBook",
            "params": [market_address]
        }
        
        response = requests.post(self.rpc, json=payload)
        data = response.json()
        
        bids = data.get("result", {}).get("bids", [])
        asks = data.get("result", {}).get("asks", [])
        
        return {
            "bids": bids[:depth],
            "asks": asks[:depth],
            "spread": float(asks[0][0]) - float(bids[0][0]) if asks and bids else 0,
            "total_bid_depth": sum(float(b[1]) for b in bids),
            "total_ask_depth": sum(float(a[1]) for a in asks)
        }
```

```python
class DeFiLlamaAggregator:
    """DeFiLlama 多链数据聚合"""
    
    BASE_URL = "https://api.llama.fi"

        """获取所有协议数据"""

        response = requests.get(url)
        return response.json()
    
    def get_protocol_tvl(self, protocol: str) -> dict:
        """获取协议 TVL"""
        url = f"{self.BASE_URL}/protocol/{protocol}"
        response = requests.get(url)
        return response.json()
    
    def get_chain_tvl(self, chain: str) -> dict:
        """获取链 TVL"""
        url = f"{self.BASE_URL}/overview/feeds/{chain}"
        response = requests.get(url)
        return response.json()
    
    def get_stablecoins_mcap(self) -> dict:
        """获取稳定币市值"""
        url = f"{self.BASE_URL}/stablecoin"
        response = requests.get(url)
        return response.json()
    
    def get_yields(self, pool: str = None) -> list:
        """获取收益数据"""
        url = f"{self.BASE_URL}/pools/{pool}" if pool else f"{self.BASE_URL}/pools"
        response = requests.get(url)
        return response.json()
    
    def cross_chain_tvl_comparison(self, chains: list) -> dict:
        """
        多链 TVL 对比
        """
        tvl_data = {}
        
        for chain in chains:
            chain_data = self.get_chain_tvl(chain)
            tvl_data[chain] = {
                "tvl": chain_data.get("tvl", 0),
                "change_1d": chain_data.get("change_1d", 0),
                "change_7d": chain_data.get("change_7d", 0),
                "top_protocols": chain_data.get("topProtocols", [])[:5]
            }
        
        # 计算占比
        total_tvl = sum(d["tvl"] for d in tvl_data.values())
        
        for chain in tvl_data:
            tvl_data[chain]["share_pct"] = tvl_data[chain]["tvl"] / total_tvl * 100 if total_tvl else 0
        
        return {
            "chain_data": tvl_data,
            "total_tvl": total_tvl,
            "leader": max(tvl_data.items(), key=lambda x: x[1]["tvl"])[0]
        }
```

## 六、数据质量与延迟

### 数据质量检查

```python
def validate_data_quality(data: dict, source: str) -> dict:
    """
    验证数据质量
    
    检查项：
    1. 数据完整性（无缺失字段）
    2. 数据时效性（更新时间）
    3. 数据一致性（跨源对比）
    """
    issues = []
    
    # 检查1：数据完整性
    required_fields = ["timestamp", "value", "source"]
    for field in required_fields:
        if field not in data:
            issues.append({"field": field, "issue": "missing"})
    
    # 检查2：数据时效性
    if "timestamp" in data:
        age_seconds = time.time() - data["timestamp"]
        if age_seconds > 3600:  # > 1 小时
            issues.append({"field": "timestamp", "issue": "stale", "age_seconds": age_seconds})
    
    # 检查3：数据合理性
    if "value" in data:
        if data["value"] < 0:
            issues.append({"field": "value", "issue": "negative_value"})
        if data["value"] > 1e12:  # > $1T（不合理）
            issues.append({"field": "value", "issue": "unrealistic_value"})
    
    return {
        "source": source,
        "quality_score": max(0, 100 - len(issues) * 20),
        "issues": issues,
        "is_valid": len(issues) == 0
    }
```

## 💡 跨链数据实战要点

### 1. L2 vs L1 选择

```
选择 L2 的场景：
  ✅ Gas 费节省 > 80%（频繁交易）
  ✅ TVL 高（流动性好）
  ✅ EVM 兼容（开发简单）
  ✅ 生态成熟（有主流协议）

选择 L1 的场景：
  ✅ 安全性优先（资金量大）
  ✅ DeFi 协议最齐全
  ✅ 需要跨链互操作性
  ✅ MEV 保护需求高
```
